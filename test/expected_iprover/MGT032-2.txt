Axiom 2: environment(sk2)
Axiom 3: environment(X) /\ stable(X) => in_environment(X,sk1(X))
Axiom 4: in_environment(sk2,X) => subpopulations(first_movers,efficient_producers,sk2,sk3(X))
Axiom 5: in_environment(sk2,X) => greater_or_equal(sk3(X),X)
Axiom 6: subpopulations(first_movers,efficient_producers,X,Y) /\ greater_or_equal(Y,sk1(X)) /\ environment(X) /\ stable(X) => greater(growth_rate(efficient_producers,Y),growth_rate(first_movers,Y))
Axiom 7: greater(growth_rate(X,Y),growth_rate(Z,Y)) /\ subpopulations(Z,X,A,Y) /\ environment(A) => selection_favors(X,Z,Y)

Lemma 8: greater_or_equal(sk3(sk1(sk2)),sk1(sk2))
