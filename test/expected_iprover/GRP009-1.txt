Axiom 1: product(identity,X,X)
Axiom 2: product(X,inverse(X),identity)
Axiom 3: product(a,b,identity)
Axiom 4: product(X,Y,Z) /\ product(Y,A,B) /\ product(Z,A,C) => product(X,B,C)
Axiom 5: product(X,identity,X)
Axiom 6: product(X,Y,Z) /\ product(X,Y,A) => Z = A
Axiom 7: product(c,b,identity)

Lemma 8: product(X,identity,Y) => X = Y
Proof:
  have product(X,identity,X)
    by axiom 5
   and product(X,identity,Y)
    by assumption
  hence X = Y
    by axiom 6

Lemma 9: a = inverse(b)
Proof:
  have product(a,b,identity)
    by axiom 3
   and product(b,inverse(b),identity)
    by axiom 2
   and product(identity,inverse(b),inverse(b))
    by axiom 1
  hence product(a,identity,inverse(b))
    by axiom 4
  hence a = inverse(b)
    by lemma 8

Lemma 10: product(b,a,identity)
Proof:
  have product(b,inverse(b),identity)
    by axiom 2
  hence product(b,a,identity)
    by rw lemma 9 R->L

Goal 1: a = c
Proof:
  have product(c,b,identity)
    by axiom 7
   and product(b,a,identity)
    by lemma 10
   and product(identity,a,a)
    by axiom 1
  hence product(c,identity,a)
    by axiom 4
  hence a = c
    by lemma 8
