Axiom 2: p(X) => f(X) = c
Axiom 3: q(b)
Axiom 4: q(Y) => a = Y

Lemma 5: f(a) = c

  have p(a)

  hence f(a) = c
    by axiom 2

Lemma 6: a = b
Proof:
  have q(b)
    by axiom 3
  hence a = b
    by axiom 4
