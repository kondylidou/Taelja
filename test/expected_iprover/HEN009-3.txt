Axiom 1: less_equal(zero,X)
Axiom 2: less_equal(X,Y) => divide(X,Y) = zero
Axiom 3: less_equal(X,identity)
Axiom 4: divide(identity,a) = b
Axiom 5: less_equal(divide(divide(X,Y),divide(Z,Y)),divide(divide(X,Z),Y))
Axiom 6: less_equal(X,Y) /\ less_equal(Y,X) => X = Y
Axiom 7: divide(X,Y) = zero => less_equal(X,Y)
Axiom 8: less_equal(divide(X,Y),X)
Axiom 9: divide(identity,b) = c
Axiom 10: divide(identity,c) = d

Lemma 11: divide(X,identity) = zero
Proof:
  have less_equal(X,identity)
    by axiom 3
  hence divide(X,identity) = zero
    by axiom 2

Lemma 12: divide(zero,X) = zero
Proof:
  have less_equal(zero,X)
    by axiom 1
  hence divide(zero,X) = zero
    by axiom 2

Lemma 13: less_equal(X,zero) => X = zero
Proof:
  have less_equal(zero,X)
    by axiom 1
   and less_equal(X,zero)
    by assumption
  hence X = zero
    by axiom 6

Lemma 14: divide(divide(X,a),b) = zero
Proof:
  have less_equal(divide(divide(X,a),divide(identity,a)),divide(divide(X,identity),a))
    by axiom 5
  hence less_equal(divide(divide(X,a),b),divide(divide(X,identity),a))
    by rw axiom 4
  hence less_equal(divide(divide(X,a),b),divide(zero,a))
    by rw lemma 11
  hence less_equal(divide(divide(X,a),b),zero)
    by rw lemma 12
  hence divide(divide(X,a),b) = zero
    by lemma 13

Lemma 15: divide(divide(X,Y),X) = zero
Proof:
  have less_equal(divide(X,Y),X)
    by axiom 8
  hence divide(divide(X,Y),X) = zero
    by axiom 2

Lemma 16: divide(divide(X,b),c) = zero

Lemma 18: less_equal(divide(a,b),c)
Proof:

    by axiom 7

Lemma 19: c = divide(a,b)
Proof:
  have less_equal(divide(divide(identity,b),divide(a,b)),divide(divide(identity,a),b))
    by axiom 5
  hence less_equal(divide(divide(identity,b),divide(a,b)),zero)
    by rw lemma 14
  hence less_equal(divide(c,divide(a,b)),zero)
    by rw axiom 9
  hence divide(c,divide(a,b)) = zero
    by lemma 13
  hence less_equal(c,divide(a,b))
    by axiom 7
   and less_equal(divide(a,b),c)
    by lemma 18
  hence c = divide(a,b)
    by axiom 6
