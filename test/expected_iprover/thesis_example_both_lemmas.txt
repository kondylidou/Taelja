Axiom 1: f(X) = g(X)
Axiom 2: f(X) = X
Axiom 3: g(X) = X => q(X)
Axiom 4: s(a)
Axiom 5: q(X) /\ s(X) => p(X)

Lemma 6: g(a) = a
Proof:
  g(a)
= { by axiom 1 R->L }
  f(a)
= { by axiom 2 }
  a

Goal 1: p(a)
Proof:
  have g(a) = a
    by lemma 6
  hence q(a)
    by axiom 3
   and s(a)
    by axiom 4
  hence p(a)
    by axiom 5
