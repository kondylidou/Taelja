Axiom 1: less_equal(divide(X,Y),X)
Axiom 2: less_equal(zero,X)
Axiom 3: less_equal(divide(a,b),d)
Axiom 4: less_equal(X,Y) => divide(X,Y) = zero
Axiom 5: less_equal(divide(divide(X,Y),divide(Z,Y)),divide(divide(X,Z),Y))
Axiom 6: less_equal(X,Y) /\ less_equal(Y,Z) => less_equal(X,Z)
Axiom 7: less_equal(X,Y) /\ less_equal(Y,X) => X = Y
Axiom 8: divide(X,Y) = zero => less_equal(X,Y)

Lemma 9: divide(divide(a,b),d) = zero
Proof:
  have less_equal(divide(a,b),d)
    by axiom 3
  hence divide(divide(a,b),d) = zero
    by axiom 4

Lemma 10: less_equal(divide(divide(a,d),divide(b,d)),zero)
Proof:
  have less_equal(divide(divide(a,d),divide(b,d)),divide(divide(a,b),d))
    by axiom 5
  hence less_equal(divide(divide(a,d),divide(b,d)),zero)
    by rw lemma 9

Goal 1: less_equal(divide(a,d),b)
Proof:
  have less_equal(zero,divide(divide(a,d),divide(b,d)))
    by axiom 2
   and less_equal(divide(divide(a,d),divide(b,d)),zero)
    by lemma 10
  hence zero = divide(divide(a,d),divide(b,d))
    by axiom 7
  hence less_equal(divide(a,d),divide(b,d))
    by axiom 8
   and less_equal(divide(b,d),b)
    by axiom 1
  hence less_equal(divide(a,d),b)
    by axiom 6
