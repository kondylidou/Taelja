Axiom 1: mult(X,ld(X,Y)) = Y
Axiom 2: mult(rd(X,Y),Y) = X
Axiom 3: mult(mult(X,mult(Y,Y)),Z) = mult(mult(X,Y),mult(Y,Z))
Axiom 4: rd(mult(X,Y),Y) = X

Lemma 5: mult(mult(rd(X,mult(Y,Y)),Y),mult(Y,Z)) = mult(X,Z)
Proof:
  mult(mult(rd(X,mult(Y,Y)),Y),mult(Y,Z))
= { by axiom 3 R->L }
  mult(mult(rd(X,mult(Y,Y)),mult(Y,Y)),Z)
= { by axiom 2 }
  mult(X,Z)

Lemma 6: mult(mult(rd(X,mult(Y,Y)),Y),Z) = mult(X,ld(Y,Z))
Proof:
  mult(mult(rd(X,mult(Y,Y)),Y),Z)
= { by axiom 1 R->L }
  mult(mult(rd(X,mult(Y,Y)),Y),mult(Y,ld(Y,Z)))
= { by lemma 5 }
  mult(X,ld(Y,Z))

Lemma 7: mult(rd(X,mult(X,X)),X) = rd(Y,Y)
Proof:
  mult(rd(X,mult(X,X)),X)
= { by axiom 4 R->L }
  rd(mult(mult(rd(X,mult(X,X)),X),Y),Y)
= { by lemma 6 }
  rd(mult(X,ld(X,Y)),Y)
= { by axiom 1 }
  rd(Y,Y)

Lemma 8: rd(X,X) = rd(Y,Y)
Proof:
  rd(X,X)
= { by lemma 7 R->L }
  mult(rd(X,mult(X,X)),X)
= { by lemma 7 }
  rd(Y,Y)

Lemma 9: mult(rd(X,X),Y) = Y
Proof:
  mult(rd(X,X),Y)
= { by lemma 8 }
  mult(rd(Y,Y),Y)
= { by axiom 2 }
  Y

Lemma 10: mult(rd(rd(X,Y),mult(Z,Z)),Z) = rd(X,mult(Z,Y))
Proof:
  mult(rd(rd(X,Y),mult(Z,Z)),Z)
= { by axiom 4 R->L }
  rd(mult(mult(rd(rd(X,Y),mult(Z,Z)),Z),mult(Z,Y)),mult(Z,Y))
= { by lemma 5 }
  rd(mult(rd(X,Y),Y),mult(Z,Y))
= { by axiom 2 }
  rd(X,mult(Z,Y))

Lemma 11: ld(X,X) = rd(Y,Y)
Proof:
  ld(X,X)
= { by lemma 9 R->L }
  mult(rd(X,X),ld(X,X))
= { by lemma 6 R->L }
  mult(mult(rd(rd(X,X),mult(X,X)),X),X)
= { by lemma 10 }
  mult(rd(X,mult(X,X)),X)
= { by lemma 7 }
  rd(Y,Y)

Goal 1: mult(rd(X,X),Y) = Y
Proof:
  mult(rd(X,X),Y)
= { by lemma 9 }
  Y

Goal 2: mult(X,rd(Y,Y)) = X
Proof:
  mult(X,rd(Y,Y))
= { by lemma 11 R->L }
  mult(X,ld(X,X))
= { by axiom 1 }
  X
