Axiom 1: s0(d)
Axiom 2: k0(e)
Axiom 3: s0(d) /\ k0(e) => q1(d,d,d)
Axiom 4: q0(X,d)
Axiom 5: q1(d,X,d) /\ q0(Y,X) /\ s0(d) => r1(Y)
Axiom 6: l0(c)
Axiom 7: l0(X) /\ r1(X) => r2(X)

Lemma 8: r1(c)
Proof:
  have s0(d)
    by axiom 1
   and k0(e)
    by axiom 2
  hence q1(d,d,d)
    by axiom 3
   and q0(c,d)
    by axiom 4
   and s0(d)
    by axiom 1
  hence r1(c)
    by axiom 5

Goal 1: r2(c)
Proof:
  have l0(c)
    by axiom 6
   and r1(c)
    by lemma 8
  hence r2(c)
    by axiom 7
