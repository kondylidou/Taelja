Axiom 1: s(a)
Axiom 2: s(X) => p(X)
Axiom 3: t(b)
Axiom 4: t(Y) => q(Y)
Axiom 5: q(Y) /\ p(X) => r(X,Y)

Lemma 6: p(a)
Proof:
  have s(a)
    by axiom 1
  hence p(a)
    by axiom 2

Goal 1: r(a,b)
Proof:
  have t(b)
    by axiom 3
  hence q(b)
    by axiom 4
   and p(a)
    by lemma 6
  hence r(a,b)
    by axiom 5
