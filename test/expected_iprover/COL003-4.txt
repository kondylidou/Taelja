Axiom 1: apply(apply(X,Y),Y) = apply(apply(w,X),Y)
Axiom 2: apply(apply(apply(b,X),Y),Z) = apply(X,apply(Y,Z))
Axiom 3: apply(fixed_pt,apply(X,fixed_pt)) = apply(X,fixed_pt) => fixed_point(X)

Lemma 4: apply(apply(X,X),X) = apply(apply(w,w),X)
Proof:
  apply(apply(X,X),X)
= { by axiom 1 }
  apply(apply(w,X),X)
= { by axiom 1 }
  apply(apply(w,w),X)

Lemma 5: apply(apply(b,fixed_pt),apply(X,X)) = apply(apply(w,apply(b,apply(b,fixed_pt))),X)
Proof:
  apply(apply(b,fixed_pt),apply(X,X))
= { by axiom 2 R->L }
  apply(apply(apply(b,apply(b,fixed_pt)),X),X)
= { by axiom 1 }
  apply(apply(w,apply(b,apply(b,fixed_pt))),X)

Lemma 6: apply(fixed_pt,apply(apply(apply(b,apply(apply(b,apply(w,w)),apply(apply(b,w),b))),b),fixed_pt)) = apply(apply(apply(b,apply(apply(b,apply(w,w)),apply(apply(b,w),b))),b),fixed_pt)
Proof:
  apply(fixed_pt,apply(apply(apply(b,apply(apply(b,apply(w,w)),apply(apply(b,w),b))),b),fixed_pt))
= { by axiom 2 }
  apply(fixed_pt,apply(apply(apply(b,apply(w,w)),apply(apply(b,w),b)),apply(b,fixed_pt)))
= { by axiom 2 }
  apply(fixed_pt,apply(apply(w,w),apply(apply(apply(b,w),b),apply(b,fixed_pt))))
= { by axiom 2 }
  apply(fixed_pt,apply(apply(w,w),apply(w,apply(b,apply(b,fixed_pt)))))
= { by lemma 4 R->L }
  apply(fixed_pt,apply(apply(apply(w,apply(b,apply(b,fixed_pt))),apply(w,apply(b,apply(b,fixed_pt)))),apply(w,apply(b,apply(b,fixed_pt)))))
= { by axiom 2 R->L }
  apply(apply(apply(b,fixed_pt),apply(apply(w,apply(b,apply(b,fixed_pt))),apply(w,apply(b,apply(b,fixed_pt))))),apply(w,apply(b,apply(b,fixed_pt))))
= { by lemma 5 }
  apply(apply(apply(w,apply(b,apply(b,fixed_pt))),apply(w,apply(b,apply(b,fixed_pt)))),apply(w,apply(b,apply(b,fixed_pt))))
= { by lemma 4 }
  apply(apply(w,w),apply(w,apply(b,apply(b,fixed_pt))))
= { by axiom 2 R->L }
  apply(apply(w,w),apply(apply(apply(b,w),b),apply(b,fixed_pt)))
= { by axiom 2 R->L }
  apply(apply(apply(b,apply(w,w)),apply(apply(b,w),b)),apply(b,fixed_pt))
= { by axiom 2 R->L }
  apply(apply(apply(b,apply(apply(b,apply(w,w)),apply(apply(b,w),b))),b),fixed_pt)

Goal 1: fixed_point(apply(apply(b,apply(apply(b,apply(w,w)),apply(apply(b,w),b))),b))
Proof:
  have apply(fixed_pt,apply(apply(apply(b,apply(apply(b,apply(w,w)),apply(apply(b,w),b))),b),fixed_pt)) = apply(apply(apply(b,apply(apply(b,apply(w,w)),apply(apply(b,w),b))),b),fixed_pt)

  hence fixed_point(apply(apply(b,apply(apply(b,apply(w,w)),apply(apply(b,w),b))),b))
