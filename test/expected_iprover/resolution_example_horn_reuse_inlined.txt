Axiom 1: q(b,X)
Axiom 2: p(X)
Axiom 3: p(Y) /\ q(Z,Y) => q(f(Z),Y)

Lemma 4: q(X,Y) => q(f(X),Y)
Proof:
  have p(Y)
    by axiom 2
   and q(X,Y)
    by assumption
  hence q(f(X),Y)
    by axiom 3

Goal 1: q(f(f(b)),a)
Proof:
  have q(b,a)
    by axiom 1
  hence q(f(b),a)
    by lemma 4
  hence q(f(f(b)),a)
    by lemma 4
