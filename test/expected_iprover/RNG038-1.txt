Axiom 1: sum(additive_identity,X,X)
Axiom 2: X = additive_identity => product(X,h(X,Y),Y)
Axiom 3: product(X,Y,Z) /\ product(A,Y,B) /\ product(C,Y,U) /\ sum(X,A,C) => sum(Z,B,U)
Axiom 4: sum(X,Y,add(X,Y))
Axiom 5: sum(X,additive_inverse(X),additive_identity)
Axiom 6: sum(X,Y,Z) /\ sum(Y,A,B) /\ sum(Z,A,C) => sum(X,B,C)
Axiom 7: sum(X,additive_identity,X)
Axiom 8: sum(X,Y,Z) /\ sum(X,Y,A) => Z = A
Axiom 9: sum(additive_inverse(X),X,additive_identity)
Axiom 10: sum(X,Y,Z) /\ sum(X,A,B) /\ sum(Y,C,A) => sum(Z,C,B)
Axiom 11: sum(X,Y,Z) => sum(Y,X,Z)

Lemma 12: product(additive_identity,h(additive_identity,X),X)
Proof:
  have product(additive_identity,h(additive_identity,X),X)
    by axiom 2

Lemma 13: sum(X,Y,add(Y,X))
Proof:
  have sum(Y,X,add(Y,X))
    by axiom 4
  hence sum(X,Y,add(Y,X))
    by axiom 11

Lemma 14: sum(additive_identity,X,add(Y,Z)) => sum(additive_identity,add(Y,Z),add(X,additive_identity))
Proof:
  have sum(additive_identity,additive_identity,additive_identity)
    by axiom 7
   and sum(additive_identity,X,add(Y,Z))
    by assumption
   and sum(additive_identity,X,add(X,additive_identity))
    by lemma 13
  hence sum(additive_identity,add(Y,Z),add(X,additive_identity))
    by axiom 6

Lemma 15: X = add(X,additive_identity)
Proof:
  have sum(additive_identity,X,X)
    by axiom 1
   and sum(additive_identity,X,add(X,additive_identity))
    by lemma 13
  hence X = add(X,additive_identity)
    by axiom 8

Lemma 16: sum(additive_identity,X,add(Y,Z)) => sum(Y,Z,X)
Proof:
  have sum(additive_identity,Y,Y)
    by axiom 1
   and sum(additive_identity,add(Y,Z),add(X,additive_identity))
    by lemma 14
   and sum(Y,Z,add(Y,Z))
    by axiom 4
  hence sum(Y,Z,add(X,additive_identity))
    by axiom 10
  hence sum(Y,Z,X)
    by rw lemma 15 R->L

Lemma 17: sum(X,additive_identity,additive_inverse(additive_inverse(X)))
Proof:
  have sum(X,additive_inverse(X),additive_identity)
    by axiom 5
   and sum(additive_inverse(X),additive_inverse(additive_inverse(X)),additive_identity)
    by axiom 5
   and sum(additive_identity,additive_inverse(additive_inverse(X)),additive_inverse(additive_inverse(X)))
    by axiom 1
  hence sum(X,additive_identity,additive_inverse(additive_inverse(X)))
    by axiom 6

Lemma 18: X = additive_inverse(additive_inverse(X))
Proof:
  have sum(X,additive_identity,X)
    by axiom 7
   and sum(X,additive_identity,additive_inverse(additive_inverse(X)))
    by lemma 17
  hence X = additive_inverse(additive_inverse(X))
    by axiom 8

Lemma 19: sum(additive_inverse(X),Y,Z) => sum(X,Z,Y)
Proof:
  have sum(additive_inverse(additive_inverse(X)),additive_inverse(X),additive_identity)
    by axiom 9
   and sum(additive_inverse(additive_inverse(X)),Z,add(additive_inverse(additive_inverse(X)),Z))
    by axiom 4
   and sum(additive_inverse(X),Y,Z)
    by assumption
  hence sum(additive_identity,Y,add(additive_inverse(additive_inverse(X)),Z))
    by axiom 10
  hence sum(additive_inverse(additive_inverse(X)),Z,Y)
    by lemma 16
  hence sum(X,Z,Y)
    by rw lemma 18 R->L

Lemma 20: sum(add(X,Y),Z,A) /\ sum(Y,Z,B) => sum(X,B,A)
Proof:
  have sum(X,Y,add(X,Y))
    by axiom 4
   and sum(Y,Z,B)
    by assumption
   and sum(add(X,Y),Z,A)
    by assumption
  hence sum(X,B,A)
    by axiom 6

Lemma 21: sum(b,b,b)
Proof:
  have product(additive_identity,h(additive_identity,b),b)
    by lemma 12
   and product(additive_identity,h(additive_identity,b),b)
    by lemma 12
   and product(additive_identity,h(additive_identity,b),b)
    by lemma 12
   and sum(additive_identity,additive_identity,additive_identity)
    by axiom 1
  hence sum(b,b,b)
    by axiom 3

Lemma 22: sum(additive_inverse(b),b,b)
Proof:
  have sum(additive_inverse(b),b,add(additive_inverse(b),b))
    by axiom 4
  hence sum(b,add(additive_inverse(b),b),b)
    by lemma 19
  hence sum(add(additive_inverse(b),b),b,b)
    by axiom 11
   and sum(b,b,b)
    by lemma 21
  hence sum(additive_inverse(b),b,b)
    by lemma 20

Lemma 23: additive_identity = add(a,additive_inverse(a))
Proof:
  have sum(additive_inverse(a),a,additive_identity)
    by axiom 9
   and sum(additive_inverse(a),a,add(a,additive_inverse(a)))
    by lemma 13
  hence additive_identity = add(a,additive_inverse(a))
    by axiom 8

Lemma 24: sum(a,additive_inverse(a),b)
Proof:
  have sum(additive_inverse(b),additive_identity,add(additive_identity,additive_inverse(b)))
    by lemma 13
  hence sum(b,add(additive_identity,additive_inverse(b)),additive_identity)
    by lemma 19
  hence sum(add(additive_identity,additive_inverse(b)),b,additive_identity)
    by axiom 11
   and sum(additive_inverse(b),b,b)
    by lemma 22
  hence sum(additive_identity,b,additive_identity)
    by lemma 20
  hence sum(additive_identity,b,add(a,additive_inverse(a)))
    by rw lemma 23
  hence sum(a,additive_inverse(a),b)
    by lemma 16

Goal 1: additive_identity = b
Proof:
  have sum(a,additive_inverse(a),additive_identity)
    by axiom 5
   and sum(a,additive_inverse(a),b)
    by lemma 24
  hence additive_identity = b
    by axiom 8
