Axiom 1: s(X)
Axiom 2: s(Y) => p(Y)
Axiom 3: s(Y) => q(b,Y)
Axiom 4: p(Z) /\ q(Y,Z) => r(Y,Z)

Lemma 5: q(b,a)
Proof:
  have s(a)
    by axiom 1
  hence q(b,a)
    by axiom 3

Goal 1: r(b,a)
Proof:
  have s(a)
    by axiom 1
  hence p(a)
    by axiom 2
   and q(b,a)
    by lemma 5
  hence r(b,a)
    by axiom 4
