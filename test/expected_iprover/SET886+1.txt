Axiom 1: unordered_pair(X,X) = singleton(X)
Axiom 2: subset(unordered_pair(sK2,sK3),singleton(sK4))
Axiom 3: subset(unordered_pair(X,Y),singleton(Z)) => X = Z
Axiom 4: unordered_pair(X,Y) = unordered_pair(Y,X)

Lemma 5: sK2 = sK4
Proof:
  have subset(unordered_pair(sK2,sK3),singleton(sK4))
    by axiom 2
  hence sK2 = sK4
    by axiom 3

Lemma 6: sK3 = sK2
Proof:
  have subset(unordered_pair(sK2,sK3),singleton(sK4))
    by axiom 2
  hence subset(unordered_pair(sK2,sK3),singleton(sK2))
    by rw lemma 5 R->L
  hence subset(unordered_pair(sK3,sK2),singleton(sK2))
    by rw axiom 4 R->L
  hence sK3 = sK2
    by axiom 3

Goal 1: unordered_pair(sK2,sK3) = singleton(sK4)
Proof:
  have unordered_pair(sK2,sK2) = singleton(sK2)
    by axiom 1
  hence unordered_pair(sK2,sK3) = singleton(sK2)
    by rw lemma 6 R->L
  hence unordered_pair(sK2,sK3) = singleton(sK4)
    by rw lemma 5
