Axiom 1: less_equal(divide(X,Y),X)
Axiom 2: less_equal(X,Y) => divide(X,Y) = zero
Axiom 3: less_equal(divide(divide(X,Y),divide(Z,Y)),divide(divide(X,Z),Y))
Axiom 4: less_equal(zero,X)
Axiom 5: less_equal(X,Y) /\ less_equal(Y,X) => X = Y
Axiom 6: divide(X,Y) = zero => less_equal(X,Y)

Lemma 7: divide(divide(X,Y),X) = zero
Proof:
  have less_equal(divide(X,Y),X)
    by axiom 1
  hence divide(divide(X,Y),X) = zero
    by axiom 2

Lemma 8: less_equal(X,zero) => X = zero
Proof:
  have less_equal(zero,X)
    by axiom 4
   and less_equal(X,zero)
    by assumption
  hence X = zero
    by axiom 5

Goal 1: divide(a,a) = zero
Proof:
  have less_equal(divide(divide(a,a),divide(divide(a,a),a)),divide(divide(a,divide(a,a)),a))
    by axiom 3
  hence less_equal(divide(divide(a,a),divide(divide(a,a),a)),zero)
    by rw lemma 7
  hence less_equal(divide(divide(a,a),zero),zero)
    by rw lemma 7
  hence divide(divide(a,a),zero) = zero
    by lemma 8
  hence less_equal(divide(a,a),zero)
    by axiom 6
  hence divide(a,a) = zero
    by lemma 8
