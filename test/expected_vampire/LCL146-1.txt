Axiom 1: implies(truth,X) = X
Axiom 2: implies(implies(X,Y),implies(implies(Y,Z),implies(X,Z))) = truth
Axiom 3: implies(implies(X,Y),Y) = implies(implies(Y,X),X)
Axiom 4: implies(implies(not(X),not(Y)),implies(Y,X)) = truth
Axiom 5: big_V(X,Y) = implies(implies(X,Y),Y)
Axiom 6: big_hat(X,Y) = not(big_V(not(X),not(Y)))

Lemma 7: truth = implies(implies(X,implies(not(Y),not(Z))),implies(X,implies(Z,Y)))
Proof:
  truth
= { by axiom 2 R->L }
  implies(implies(X,implies(not(Y),not(Z))),implies(implies(implies(not(Y),not(Z)),implies(Z,Y)),implies(X,implies(Z,Y))))
= { by axiom 4 }
  implies(implies(X,implies(not(Y),not(Z))),implies(truth,implies(X,implies(Z,Y))))
= { by axiom 1 }
  implies(implies(X,implies(not(Y),not(Z))),implies(X,implies(Z,Y)))

Lemma 8: implies(implies(X,truth),truth) = implies(X,X)
Proof:
  implies(implies(X,truth),truth)
= { by axiom 3 }
  implies(implies(truth,X),X)
= { by axiom 1 }
  implies(X,X)

Lemma 9: truth = implies(X,X)
Proof:
  truth
= { by axiom 2 R->L }

= { by axiom 1 }

= { by axiom 1 }
  implies(truth,implies(implies(truth,X),X))
= { by axiom 1 }
  implies(implies(truth,X),X)
= { by axiom 1 }
  implies(X,X)

Lemma 10: truth = implies(X,truth)
Proof:
  truth

Goal 1: not(big_hat(x,y)) = big_V(not(x),not(y))
Proof:
  not(big_hat(x,y))
= { by axiom 6 }
  not(not(big_V(not(x),not(y))))
= { by axiom 5 }
  not(not(implies(implies(not(x),not(y)),not(y))))
= { by axiom 3 }
  not(not(implies(implies(not(y),not(x)),not(x))))
= { by axiom 1 R->L }
  implies(truth,not(not(implies(implies(not(y),not(x)),not(x)))))
= { by lemma 14 }
  implies(implies(implies(implies(not(y),not(x)),not(x)),not(not(implies(implies(not(y),not(x)),not(x))))),not(not(implies(implies(not(y),not(x)),not(x)))))
= { by axiom 3 R->L }
  implies(implies(not(not(implies(implies(not(y),not(x)),not(x)))),implies(implies(not(y),not(x)),not(x))),implies(implies(not(y),not(x)),not(x)))
= { by lemma 13 R->L }
  implies(truth,implies(implies(not(y),not(x)),not(x)))
= { by axiom 1 }
  implies(implies(not(y),not(x)),not(x))
= { by axiom 3 R->L }
  implies(implies(not(x),not(y)),not(y))
= { by axiom 5 R->L }
  big_V(not(x),not(y))
