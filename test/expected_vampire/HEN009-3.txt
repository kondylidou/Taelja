Axiom 1: less_equal(zero,X)
Axiom 2: less_equal(X,Y) => divide(X,Y) = zero
Axiom 3: less_equal(X,identity)
Axiom 4: divide(identity,a) = b
Axiom 5: less_equal(divide(divide(X,Y),divide(Z,Y)),divide(divide(X,Z),Y))
Axiom 6: less_equal(Y,X) /\ less_equal(X,Y) => X = Y
Axiom 7: divide(X,Y) = zero => less_equal(X,Y)
Axiom 8: divide(identity,b) = c
Axiom 9: less_equal(divide(X,Y),X)
Axiom 10: divide(identity,c) = d

Lemma 11: zero = divide(zero,X)
Proof:
  have less_equal(zero,X)
    by axiom 1
  hence zero = divide(zero,X)
    by axiom 2

Lemma 12: zero = divide(X,identity)
Proof:
  have less_equal(X,identity)
    by axiom 3
  hence zero = divide(X,identity)
    by axiom 2

Lemma 13: less_equal(divide(divide(X,a),b),zero)
Proof:
  less_equal(divide(divide(X,a),b),zero)
= { by lemma 11 }
  less_equal(divide(divide(X,a),b),divide(zero,a))
= { by lemma 12 }
  less_equal(divide(divide(X,a),b),divide(divide(X,identity),a))
= { by axiom 4 R->L }
  less_equal(divide(divide(X,a),divide(identity,a)),divide(divide(X,identity),a))
= { by axiom 5 }
  true

Lemma 14: less_equal(divide(divide(X,b),c),zero)
Proof:

  true

Lemma 15: less_equal(divide(b,b),zero)
Proof:
  less_equal(divide(b,b),zero)
= { by axiom 4 R->L }
  less_equal(divide(divide(identity,a),b),zero)
= { by lemma 13 }
  true

Lemma 16: divide(b,b) = zero
Proof:
  have less_equal(zero,divide(b,b))
    by axiom 1
   and less_equal(divide(b,b),zero)
    by lemma 15
  hence divide(b,b) = zero
    by axiom 6

Lemma 17: less_equal(divide(c,divide(a,b)),zero)
Proof:
  less_equal(divide(c,divide(a,b)),zero)
= { by lemma 16 R->L }

  less_equal(divide(c,divide(a,b)),divide(divide(identity,a),b))
= { by axiom 8 R->L }
  less_equal(divide(divide(identity,b),divide(a,b)),divide(divide(identity,a),b))
= { by axiom 5 }
  true

Lemma 18: less_equal(divide(a,b),c)

Lemma 19: less_equal(divide(b,divide(c,a)),divide(d,a))

Lemma 20: divide(divide(a,b),a) = zero

Lemma 21: divide(a,b) = c
Proof:
  have less_equal(zero,divide(c,divide(a,b)))
    by axiom 1
   and less_equal(divide(c,divide(a,b)),zero)

  hence divide(c,divide(a,b)) = zero

    by lemma 18

    by axiom 6

Lemma 31: less_equal(divide(b,divide(d,a)),zero)
Proof:
  have less_equal(divide(divide(b,divide(d,a)),divide(zero,divide(d,a))),divide(divide(b,zero),divide(d,a)))
    by axiom 5
   and less_equal(divide(divide(b,zero),divide(d,a)),divide(divide(b,divide(d,a)),divide(zero,divide(d,a))))

  hence divide(divide(b,zero),divide(d,a)) = divide(divide(b,divide(d,a)),divide(zero,divide(d,a)))

  hence zero = divide(divide(b,divide(d,a)),divide(zero,divide(d,a)))
    by rw lemma 30 R->L
  hence zero = divide(divide(b,divide(d,a)),zero)
