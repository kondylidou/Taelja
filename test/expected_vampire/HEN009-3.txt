Axiom 1: less_equal(zero,X)
Axiom 2: less_equal(X,Y) => divide(X,Y) = zero
Axiom 3: less_equal(X,identity)
Axiom 4: divide(identity,a) = b
Axiom 5: less_equal(divide(divide(X,Y),divide(Z,Y)),divide(divide(X,Z),Y))
Axiom 6: less_equal(Y,X) /\ less_equal(X,Y) => X = Y
Axiom 7: divide(X,Y) = zero => less_equal(X,Y)
Axiom 8: divide(identity,b) = c
Axiom 9: less_equal(divide(X,Y),X)
Axiom 10: divide(identity,c) = d

Lemma 11: zero = divide(X,identity)
Proof:
  have less_equal(X,identity)
    by axiom 3
  hence zero = divide(X,identity)
    by axiom 2

Lemma 12: zero = divide(zero,X)
Proof:
  have less_equal(zero,X)
    by axiom 1
  hence zero = divide(zero,X)
    by axiom 2

Lemma 13: less_equal(divide(divide(X,a),b),zero)
Proof:
  have less_equal(divide(divide(X,a),divide(identity,a)),divide(divide(X,identity),a))
    by axiom 5
  hence less_equal(divide(divide(X,a),b),divide(divide(X,identity),a))
    by rw axiom 4
  hence less_equal(divide(divide(X,a),b),divide(zero,a))
    by rw lemma 11 R->L
  hence less_equal(divide(divide(X,a),b),zero)
    by rw lemma 12 R->L

Lemma 14: less_equal(X,zero) => zero = X
Proof:
  have less_equal(zero,X)
    by axiom 1
   and less_equal(X,zero)
    by assumption
  hence zero = X
    by axiom 6

Lemma 15: less_equal(divide(divide(X,b),c),zero)
Proof:
  have less_equal(divide(divide(X,b),divide(identity,b)),divide(divide(X,identity),b))
    by axiom 5
  hence less_equal(divide(divide(X,b),c),divide(divide(X,identity),b))
    by rw axiom 8
  hence less_equal(divide(divide(X,b),c),divide(zero,b))
    by rw lemma 11 R->L
  hence less_equal(divide(divide(X,b),c),zero)
    by rw lemma 12 R->L

Lemma 16: zero = divide(b,b)
Proof:
  have less_equal(divide(divide(identity,a),b),zero)
    by lemma 13
  hence less_equal(divide(b,b),zero)
    by rw axiom 4
  hence zero = divide(b,b)
    by lemma 14

Lemma 17: less_equal(divide(a,b),c)
Proof:
  have less_equal(divide(divide(a,b),c),zero)
    by lemma 15
  hence zero = divide(divide(a,b),c)
    by lemma 14
  hence less_equal(divide(a,b),c)
    by axiom 7

Lemma 18: divide(divide(a,b),a) = zero
Proof:
  have less_equal(divide(a,b),a)
    by axiom 9
  hence divide(divide(a,b),a) = zero
    by axiom 2

Lemma 19: divide(a,b) = c
Proof:
  have less_equal(divide(divide(identity,b),divide(a,b)),divide(divide(identity,a),b))
    by axiom 5
  hence less_equal(divide(c,divide(a,b)),divide(divide(identity,a),b))
    by rw axiom 8
  hence less_equal(divide(c,divide(a,b)),divide(b,b))
    by rw axiom 4
  hence less_equal(divide(c,divide(a,b)),zero)
    by rw lemma 16 R->L
  hence zero = divide(c,divide(a,b))
    by lemma 14
  hence less_equal(c,divide(a,b))
    by axiom 7
   and less_equal(divide(a,b),c)
    by lemma 17
  hence divide(a,b) = c
    by axiom 6

Lemma 20: zero = divide(c,a)
Proof:
  zero
= { by lemma 18 R->L }
  divide(divide(a,b),a)
= { by lemma 19 }
  divide(c,a)

Lemma 21: divide(divide(b,divide(c,a)),divide(d,a)) = zero
Proof:
  have less_equal(divide(divide(identity,a),divide(c,a)),divide(divide(identity,c),a))
    by axiom 5
  hence less_equal(divide(b,divide(c,a)),divide(divide(identity,c),a))
    by rw axiom 4
  hence less_equal(divide(b,divide(c,a)),divide(d,a))
    by rw axiom 10
  hence divide(divide(b,divide(c,a)),divide(d,a)) = zero
    by axiom 2

Lemma 22: less_equal(divide(divide(b,zero),divide(d,a)),divide(divide(b,divide(d,a)),divide(zero,divide(d,a))))
Proof:
  have less_equal(zero,divide(divide(b,divide(d,a)),zero))
    by axiom 1
  hence less_equal(divide(divide(b,divide(c,a)),divide(d,a)),divide(divide(b,divide(d,a)),zero))
    by rw lemma 21 R->L
  hence less_equal(divide(divide(b,zero),divide(d,a)),divide(divide(b,divide(d,a)),zero))
    by rw lemma 20 R->L
  hence less_equal(divide(divide(b,zero),divide(d,a)),divide(divide(b,divide(d,a)),divide(zero,divide(d,a))))
    by rw lemma 12

Lemma 23: less_equal(divide(d,a),b)
Proof:

  hence zero = divide(divide(d,a),b)
    by lemma 14
  hence less_equal(divide(d,a),b)

Lemma 24: zero = divide(c,c)
Proof:
  have less_equal(divide(divide(identity,b),c),zero)
    by lemma 15
  hence less_equal(divide(c,c),zero)
    by rw axiom 8
  hence zero = divide(c,c)
    by lemma 14

Lemma 25: less_equal(divide(b,c),d)
Proof:
  have less_equal(divide(divide(b,c),divide(identity,c)),divide(divide(b,identity),c))
    by axiom 5
  hence less_equal(divide(divide(b,c),d),divide(divide(b,identity),c))
    by rw axiom 10
  hence less_equal(divide(divide(b,c),d),divide(zero,c))
    by rw lemma 11 R->L
  hence less_equal(divide(divide(b,c),d),zero)
    by rw lemma 12 R->L
  hence zero = divide(divide(b,c),d)
    by lemma 14
  hence less_equal(divide(b,c),d)
    by axiom 7

Lemma 26: divide(d,a) = b
Proof:
  have less_equal(divide(divide(b,divide(d,a)),divide(zero,divide(d,a))),divide(divide(b,zero),divide(d,a)))
    by axiom 5
   and less_equal(divide(divide(b,zero),divide(d,a)),divide(divide(b,divide(d,a)),divide(zero,divide(d,a))))
    by lemma 22
  hence divide(divide(b,zero),divide(d,a)) = divide(divide(b,divide(d,a)),divide(zero,divide(d,a)))
    by axiom 6
  hence divide(divide(b,zero),divide(d,a)) = divide(divide(b,divide(d,a)),zero)
    by rw lemma 12 R->L
  hence divide(divide(b,divide(c,a)),divide(d,a)) = divide(divide(b,divide(d,a)),zero)
    by rw lemma 20
  hence zero = divide(divide(b,divide(d,a)),zero)
    by rw lemma 21
  hence less_equal(divide(b,divide(d,a)),zero)
    by axiom 7
  hence zero = divide(b,divide(d,a))
    by lemma 14
  hence less_equal(b,divide(d,a))
    by axiom 7
   and less_equal(divide(d,a),b)
    by lemma 23
  hence divide(d,a) = b
    by axiom 6

Lemma 27: divide(b,c) = d
Proof:
  have less_equal(divide(divide(identity,c),divide(b,c)),divide(divide(identity,b),c))
    by axiom 5
  hence less_equal(divide(d,divide(b,c)),divide(divide(identity,b),c))
    by rw axiom 10
  hence less_equal(divide(d,divide(b,c)),divide(c,c))
    by rw axiom 8
  hence less_equal(divide(d,divide(b,c)),zero)
    by rw lemma 24 R->L
  hence zero = divide(d,divide(b,c))
    by lemma 14
  hence less_equal(d,divide(b,c))
    by axiom 7
   and less_equal(divide(b,c),d)
    by lemma 25
  hence divide(b,c) = d
    by axiom 6

Lemma 28: less_equal(b,divide(b,c))
Proof:
  have less_equal(divide(d,a),d)
    by axiom 9
  hence less_equal(b,d)
    by rw lemma 26
  hence less_equal(b,divide(b,c))
    by rw lemma 27 R->L

Goal 1: b = d
Proof:
  have less_equal(divide(b,c),b)
    by axiom 9
   and less_equal(b,divide(b,c))
    by lemma 28
  hence b = divide(b,c)
    by axiom 6
  hence b = d
    by rw lemma 27
