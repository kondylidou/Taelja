Axiom 1: less_equal(divide(X,Y),X)
Axiom 2: less_equal(X,Y) => divide(X,Y) = zero
Axiom 3: less_equal(divide(divide(X,Y),divide(Z,Y)),divide(divide(X,Z),Y))
Axiom 4: less_equal(zero,X)
Axiom 5: less_equal(Y,X) /\ less_equal(X,Y) => X = Y
Axiom 6: divide(X,Y) = zero => less_equal(X,Y)

Lemma 7: zero = divide(divide(X,Y),X)
Proof:
  have less_equal(divide(X,Y),X)
    by axiom 1
  hence zero = divide(divide(X,Y),X)
    by axiom 2

Lemma 8: less_equal(divide(divide(a,a),zero),zero)
Proof:
  less_equal(divide(divide(a,a),zero),zero)
= { by lemma 7 }
  less_equal(divide(divide(a,a),divide(divide(a,Z),a)),zero)
= { by lemma 7 }
  less_equal(divide(divide(a,a),divide(divide(a,Z),a)),divide(divide(a,divide(a,Z)),a))
= { by axiom 3 }
  true

Lemma 9: less_equal(divide(a,a),zero)
Proof:
  have less_equal(zero,divide(divide(a,a),zero))
    by axiom 4
   and less_equal(divide(divide(a,a),zero),zero)
    by lemma 8
  hence divide(divide(a,a),zero) = zero
    by axiom 5
  hence less_equal(divide(a,a),zero)
    by axiom 6

Goal 1: divide(a,a) = zero
Proof:
  have less_equal(zero,divide(a,a))
    by axiom 4
   and less_equal(divide(a,a),zero)
    by lemma 9
  hence divide(a,a) = zero
    by axiom 5
