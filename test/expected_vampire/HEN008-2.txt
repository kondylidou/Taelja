Axiom 1: less_equal(a,b)
Axiom 2: less_equal(X,Y) => quotient(X,Y,zero)
Axiom 3: quotient(X,Y,divide(X,Y))
Axiom 4: quotient(a,c,aQc)
Axiom 5: quotient(b,c,bQc)
Axiom 6: quotient(zero,X,zero)
Axiom 7: quotient(Z,A,B) /\ quotient(Y,C,A) /\ quotient(X,C,Z) /\ quotient(X,Y,U) /\ quotient(U,C,V) => less_equal(B,V)
Axiom 8: quotient(X,zero,X)
Axiom 9: quotient(X,C,A) /\ less_equal(U,C) /\ quotient(X,Y,U) => less_equal(A,Y)

Lemma 10: quotient(a,b,zero)
Proof:
  have less_equal(a,b)
    by axiom 1
  hence quotient(a,b,zero)
    by axiom 2

Lemma 11: less_equal(divide(aQc,bQc),zero)
Proof:
  have quotient(aQc,bQc,divide(aQc,bQc))
    by axiom 3
   and quotient(b,c,bQc)
    by axiom 5
   and quotient(a,c,aQc)
    by axiom 4
   and quotient(a,b,zero)
    by lemma 10
   and quotient(zero,c,zero)
    by axiom 6
  hence less_equal(divide(aQc,bQc),zero)
    by axiom 7

Goal 1: less_equal(aQc,bQc)
Proof:
  have quotient(aQc,zero,aQc)
    by axiom 8
   and less_equal(divide(aQc,bQc),zero)
    by lemma 11
   and quotient(aQc,bQc,divide(aQc,bQc))
    by axiom 3
  hence less_equal(aQc,bQc)
    by axiom 9
