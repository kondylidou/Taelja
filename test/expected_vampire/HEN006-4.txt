Axiom 1: less_equal(divide(a,b),d)
Axiom 2: less_equal(X,Y) => divide(X,Y) = zero
Axiom 3: less_equal(divide(divide(X,Y),divide(Z,Y)),divide(divide(X,Z),Y))
Axiom 4: less_equal(zero,X)
Axiom 5: less_equal(Y,X) /\ less_equal(X,Y) => X = Y
Axiom 6: divide(X,Y) = zero => less_equal(X,Y)
Axiom 7: less_equal(divide(X,Y),X)
Axiom 8: less_equal(Y,Z) /\ less_equal(X,Y) => less_equal(X,Z)

Lemma 9: divide(divide(a,b),d) = zero
Proof:
  have less_equal(divide(a,b),d)
    by axiom 1
  hence divide(divide(a,b),d) = zero
    by axiom 2

Lemma 10: less_equal(divide(divide(a,d),divide(b,d)),zero)
Proof:
  less_equal(divide(divide(a,d),divide(b,d)),zero)
= { by lemma 9 R->L }
  less_equal(divide(divide(a,d),divide(b,d)),divide(divide(a,b),d))
= { by axiom 3 }
  true

Lemma 11: zero = divide(divide(a,d),divide(b,d))
Proof:
  have less_equal(divide(divide(a,d),divide(b,d)),zero)
    by lemma 10
   and less_equal(zero,divide(divide(a,d),divide(b,d)))
    by axiom 4
  hence zero = divide(divide(a,d),divide(b,d))
    by axiom 5

Lemma 12: less_equal(divide(a,d),divide(b,d))
Proof:
  have zero = divide(divide(a,d),divide(b,d))
    by lemma 11
  hence less_equal(divide(a,d),divide(b,d))
    by axiom 6

Goal 1: less_equal(divide(a,d),b)
Proof:
  have less_equal(divide(a,d),divide(b,d))
    by lemma 12
   and less_equal(divide(b,d),b)
    by axiom 7
  hence less_equal(divide(a,d),b)
    by axiom 8
