Axiom 1: unordered_pair(X,X) = singleton(X)
Axiom 2: subset(unordered_pair(sK2,sK3),singleton(sK4))
Axiom 3: subset(unordered_pair(X,Y),singleton(Z)) => X = Z
Axiom 4: unordered_pair(X,Y) = unordered_pair(Y,X)

Lemma 5: subset(unordered_pair(sK2,sK3),unordered_pair(sK4,sK4))
Proof:
  have subset(unordered_pair(sK2,sK3),singleton(sK4))
    by axiom 2
  hence subset(unordered_pair(sK2,sK3),unordered_pair(sK4,sK4))
    by rw axiom 1 R->L

Lemma 6: sK2 = sK4
Proof:
  have subset(unordered_pair(sK2,sK3),unordered_pair(sK4,sK4))
    by lemma 5
  hence subset(unordered_pair(sK2,sK3),singleton(sK4))
    by rw axiom 1
  hence sK2 = sK4
    by axiom 3

Lemma 8: sK3 = sK4
Proof:
  have subset(unordered_pair(sK3,sK4),singleton(sK4))
    by lemma 7
  hence sK3 = sK4

Goal 1: unordered_pair(sK2,sK3) = singleton(sK4)
Proof:
  have unordered_pair(sK4,sK4) = unordered_pair(sK4,sK4)
    by axiom 4
