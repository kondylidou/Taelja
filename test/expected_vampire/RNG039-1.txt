Axiom 1: add(X,X) = additive_identity
Axiom 2: multiply(X,X) = X
Axiom 3: product(a,multiply(b,X),multiply(Y,X))
Axiom 4: product(a,b,c)
Axiom 5: product(X,Y,Z) /\ product(X,Y,A) => A = Z
Axiom 6: product(add(a,b),b,add(c,b))
Axiom 7: product(X,Y,multiply(X,Y))
Axiom 8: multiply(b,a) = d
Axiom 9: product(multiply(X,b),a,multiply(X,d))
Axiom 10: product(b,a,d)
Axiom 11: add(X,additive_identity) = X
Axiom 12: product(add(a,b),a,add(a,d))

Lemma 13: product(a,b,multiply(X,b))
Proof:
  have product(a,multiply(b,b),multiply(X,b))
    by axiom 3
  hence product(a,b,multiply(X,b))
    by rw axiom 2

Lemma 14: product(a,b,b)
Proof:
  have product(a,b,multiply(b,b))
    by lemma 13
  hence product(a,b,b)
    by rw axiom 2

Lemma 15: b = c
Proof:
  have product(a,b,c)
    by axiom 4
   and product(a,b,b)
    by lemma 14
  hence b = c
    by axiom 5

Lemma 16: b = multiply(X,b)
Proof:
  have product(a,b,c)
    by axiom 4
   and product(a,b,multiply(X,b))
    by lemma 13
  hence multiply(X,b) = c
    by axiom 5
  hence multiply(X,b) = b
    by rw lemma 15 R->L

Lemma 17: product(add(a,b),b,additive_identity)
Proof:
  have product(add(a,b),b,add(c,b))
    by axiom 6
  hence product(add(a,b),b,add(b,b))
    by rw lemma 15 R->L
  hence product(add(a,b),b,additive_identity)
    by rw axiom 1

Lemma 18: additive_identity = b
Proof:
  have product(add(a,b),b,multiply(add(a,b),b))
    by axiom 7
  hence product(add(a,b),b,b)
    by rw lemma 16 R->L
   and product(add(a,b),b,additive_identity)
    by lemma 17
  hence additive_identity = b
    by axiom 5

Lemma 19: product(b,a,multiply(a,d))
Proof:
  have product(multiply(a,b),a,multiply(a,d))
    by axiom 9
  hence product(b,a,multiply(a,d))
    by rw lemma 16 R->L

Lemma 20: multiply(a,d) = d
Proof:
  have product(b,a,d)
    by axiom 10
   and product(b,a,multiply(a,d))
    by lemma 19
  hence multiply(a,d) = d
    by axiom 5

Lemma 21: product(a,d,a)
Proof:
  have product(a,multiply(b,a),multiply(a,a))
    by axiom 3
  hence product(a,d,multiply(a,a))
    by rw axiom 8
  hence product(a,d,a)
    by rw axiom 2

Lemma 22: a = d
Proof:
  have product(a,d,multiply(a,d))
    by axiom 7
  hence product(a,d,d)
    by rw lemma 20
   and product(a,d,a)
    by lemma 21
  hence a = d
    by axiom 5

Lemma 23: product(b,a,multiply(a,d))
Proof:
  have product(multiply(a,b),a,multiply(a,d))
    by axiom 9
  hence product(b,a,multiply(a,d))
    by rw lemma 16 R->L

Lemma 24: multiply(a,d) = d
Proof:
  have product(b,a,d)
    by axiom 10
   and product(b,a,multiply(a,d))
    by lemma 23
  hence multiply(a,d) = d
    by axiom 5

Lemma 25: add(X,b) = X
Proof:
  add(X,b)
= { by lemma 18 R->L }
  add(X,additive_identity)
= { by axiom 11 }
  X

Lemma 26: product(a,d,b)
Proof:
  have product(add(a,b),a,add(a,d))
    by axiom 12
  hence product(a,a,add(a,d))
    by rw lemma 25
  hence product(a,a,add(a,a))
    by rw lemma 22 R->L
  hence product(a,a,additive_identity)
    by rw axiom 1
  hence product(a,a,b)
    by rw lemma 18
  hence product(a,d,b)
    by rw lemma 22

Goal 1: c = d
Proof:
  have product(a,d,multiply(a,d))
    by axiom 7
  hence product(a,d,d)
    by rw lemma 24
   and product(a,d,b)
    by lemma 26
  hence b = d
    by axiom 5
  hence c = d
    by rw lemma 15
