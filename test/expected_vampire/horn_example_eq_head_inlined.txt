Axiom 1: f(X) = g(X)
Axiom 2: p(f(X))
Axiom 3: q(f(X))
Axiom 4: p(X) /\ q(X) => X = zero

Lemma 5: q(g(a))
Proof:
  have q(f(a))
    by axiom 3
  hence q(g(a))
    by rw axiom 1

Goal 1: g(a) = zero
Proof:
  have p(f(a))
    by axiom 2
  hence p(g(a))
    by rw axiom 1
   and q(g(a))
    by lemma 5
  hence g(a) = zero
    by axiom 4
