Axiom 4: ld(X,mult(X,Y)) = Y
Axiom 5: rd(mult(X,Y),Y) = X

Lemma 6: mult(ld(X,X),Y) = Y
Proof:
  mult(ld(X,X),Y)
= { by axiom 4 R->L }
  ld(mult(Y,X),mult(mult(Y,X),mult(ld(X,X),Y)))
= { by axiom 3 }
  ld(mult(Y,X),mult(mult(Y,mult(X,ld(X,X))),Y))
= { by axiom 2 }
  ld(mult(Y,X),mult(mult(Y,X),Y))
= { by axiom 4 }
  Y
