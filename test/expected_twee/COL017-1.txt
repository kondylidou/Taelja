Axiom 1: apply(apply(apply(b,X),Y),Z) = apply(X,apply(Y,Z))
Axiom 2: apply(m,A) = apply(A,A)

Goal 1: apply(m,apply(apply(b,combinator),m)) = apply(combinator,apply(m,apply(apply(b,combinator),m)))
Proof:
  apply(m,apply(apply(b,combinator),m))
= { by axiom 2 }
  apply(apply(apply(b,combinator),m),apply(apply(b,combinator),m))
= { by axiom 1 }
  apply(combinator,apply(m,apply(apply(b,combinator),m)))
