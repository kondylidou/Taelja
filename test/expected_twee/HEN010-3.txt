Axiom 1: less_equal(zero,X)
Axiom 2: less_equal(divide(divide(X,Y),divide(Z,Y)),divide(divide(X,Z),Y))
Axiom 3: less_equal(divide(A,B),A)
Axiom 4: less_equal(X,Z) => divide(X,Z) = zero
Axiom 5: less_equal(X,Z) /\ less_equal(Z,X) => X = Z
Axiom 6: divide(X,Z) = zero => less_equal(X,Z)

Lemma 7: divide(divide(X,Y),X) = zero
Proof:
  have less_equal(divide(X,Y),X)
    by axiom 3
  hence divide(divide(X,Y),X) = zero
    by axiom 4

Lemma 8: divide(zero,X) = zero
Proof:
  have less_equal(zero,X)
    by axiom 1
  hence divide(zero,X) = zero
    by axiom 4

Lemma 9: less_equal(divide(divide(X,X),divide(zero,X)),zero)
Proof:
  have less_equal(divide(divide(X,X),divide(zero,X)),divide(divide(X,zero),X))
    by axiom 2
  hence less_equal(divide(divide(X,X),divide(zero,X)),zero)
    by rw lemma 7

Lemma 10: less_equal(divide(X,X),zero)
Proof:
  have less_equal(zero,divide(divide(X,X),divide(zero,X)))
    by axiom 1
   and less_equal(divide(divide(X,X),divide(zero,X)),zero)
    by lemma 9
  hence zero = divide(divide(X,X),divide(zero,X))
    by axiom 5
  hence zero = divide(divide(X,X),zero)
    by rw lemma 8
  hence less_equal(divide(X,X),zero)
    by axiom 6

Lemma 11: divide(X,X) = zero
Proof:
  have less_equal(zero,divide(X,X))
    by axiom 1
   and less_equal(divide(X,X),zero)
    by lemma 10
  hence divide(X,X) = zero
    by axiom 5

Lemma 12: divide(divide(divide(X,Y),zero),divide(divide(X,Y),Y)) = zero

Lemma 13: less_equal(divide(divide(divide(X,Y),divide(divide(X,Y),Y)),divide(zero,divide(divide(X,Y),Y))),zero)

Lemma 14: divide(divide(X,Y),Y) = divide(X,Y)
Proof:
  have less_equal(zero,divide(divide(divide(X,Y),divide(divide(X,Y),Y)),divide(zero,divide(divide(X,Y),Y))))
    by axiom 1
   and less_equal(divide(divide(divide(X,Y),divide(divide(X,Y),Y)),divide(zero,divide(divide(X,Y),Y))),zero)
    by lemma 13
  hence zero = divide(divide(divide(X,Y),divide(divide(X,Y),Y)),divide(zero,divide(divide(X,Y),Y)))
    by axiom 5
  hence zero = divide(divide(divide(X,Y),divide(divide(X,Y),Y)),zero)
    by rw lemma 8
  hence less_equal(divide(divide(X,Y),divide(divide(X,Y),Y)),zero)
    by axiom 6
   and less_equal(zero,divide(divide(X,Y),divide(divide(X,Y),Y)))
    by axiom 1
  hence divide(divide(X,Y),divide(divide(X,Y),Y)) = zero
    by axiom 5
  hence less_equal(divide(X,Y),divide(divide(X,Y),Y))
    by axiom 6
   and less_equal(divide(divide(X,Y),Y),divide(X,Y))
    by axiom 3
  hence divide(divide(X,Y),Y) = divide(X,Y)
    by axiom 5

Lemma 15: less_equal(divide(divide(X,divide(X,zero)),zero),divide(divide(X,zero),divide(X,zero)))

Lemma 16: divide(X,zero) = X
Proof:
  have less_equal(zero,divide(divide(X,divide(X,zero)),zero))
    by axiom 1
  hence less_equal(divide(divide(X,zero),divide(X,zero)),divide(divide(X,divide(X,zero)),zero))
    by rw lemma 11 R->L
   and less_equal(divide(divide(X,divide(X,zero)),zero),divide(divide(X,zero),divide(X,zero)))
    by lemma 15
  hence divide(divide(X,zero),divide(X,zero)) = divide(divide(X,divide(X,zero)),zero)
    by axiom 5
  hence zero = divide(divide(X,divide(X,zero)),zero)
    by rw lemma 11
  hence less_equal(divide(X,divide(X,zero)),zero)
    by axiom 6
   and less_equal(zero,divide(X,divide(X,zero)))
    by axiom 1
  hence divide(X,divide(X,zero)) = zero
    by axiom 5
  hence less_equal(X,divide(X,zero))
    by axiom 6
   and less_equal(divide(X,zero),X)
    by axiom 3
  hence divide(X,zero) = X
    by axiom 5
