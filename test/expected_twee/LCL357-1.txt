Axiom 1: is_a_theorem(implies(implies(X,Y),implies(implies(Y,Z),implies(X,Z))))
Axiom 2: is_a_theorem(implies(A,B)) /\ is_a_theorem(A) => is_a_theorem(B)

Lemma 3: is_a_theorem(implies(implies(implies(implies(y,z),implies(x,z)),implies(implies(implies(x,z),u),implies(implies(y,z),u))),implies(implies(x,y),implies(implies(implies(x,z),u),implies(implies(y,z),u)))))
Proof:
  have is_a_theorem(implies(implies(implies(x,y),implies(implies(y,z),implies(x,z))),implies(implies(implies(implies(y,z),implies(x,z)),implies(implies(implies(x,z),u),implies(implies(y,z),u))),implies(implies(x,y),implies(implies(implies(x,z),u),implies(implies(y,z),u))))))
    by axiom 1
   and is_a_theorem(implies(implies(x,y),implies(implies(y,z),implies(x,z))))
    by axiom 1
  hence is_a_theorem(implies(implies(implies(implies(y,z),implies(x,z)),implies(implies(implies(x,z),u),implies(implies(y,z),u))),implies(implies(x,y),implies(implies(implies(x,z),u),implies(implies(y,z),u)))))
    by axiom 2

Lemma 4: is_a_theorem(implies(implies(implies(y,z),implies(x,z)),implies(implies(implies(x,z),u),implies(implies(y,z),u))))
Proof:
  have is_a_theorem(implies(implies(implies(y,z),implies(x,z)),implies(implies(implies(x,z),u),implies(implies(y,z),u))))
    by axiom 1

Goal 1: is_a_theorem(implies(implies(x,y),implies(implies(implies(x,z),u),implies(implies(y,z),u))))
Proof:
  have is_a_theorem(implies(implies(implies(implies(y,z),implies(x,z)),implies(implies(implies(x,z),u),implies(implies(y,z),u))),implies(implies(x,y),implies(implies(implies(x,z),u),implies(implies(y,z),u)))))
    by lemma 3
   and is_a_theorem(implies(implies(implies(y,z),implies(x,z)),implies(implies(implies(x,z),u),implies(implies(y,z),u))))
    by lemma 4
  hence is_a_theorem(implies(implies(x,y),implies(implies(implies(x,z),u),implies(implies(y,z),u))))
    by axiom 2
