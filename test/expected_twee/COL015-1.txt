Axiom 1: apply(apply(apply(q,X),Y),Z) = apply(Y,apply(X,Z))
Axiom 2: apply(m,A) = apply(A,A)

Goal 1: apply(m,apply(apply(q,m),combinator)) = apply(combinator,apply(m,apply(apply(q,m),combinator)))
Proof:
  apply(m,apply(apply(q,m),combinator))
= { by axiom 2 }
  apply(apply(apply(q,m),combinator),apply(apply(q,m),combinator))
= { by axiom 1 }
  apply(combinator,apply(m,apply(apply(q,m),combinator)))
