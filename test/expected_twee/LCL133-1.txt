Axiom 1: implies(truth,X) = X
Axiom 2: implies(Y,Z) = implies(Z,Y)
Axiom 3: implies(implies(Y,A),A) = implies(implies(A,Y),Y)
Axiom 4: implies(implies(Y,A),implies(implies(A,B),implies(Y,B))) = truth

Lemma 5: implies(X,implies(X,Y)) = truth
Proof:
  implies(X,implies(X,Y))
= { by axiom 1 R->L }
  implies(truth,implies(X,implies(X,Y)))
= { by axiom 1 R->L }
  implies(truth,implies(truth,implies(X,implies(X,Y))))
= { by axiom 1 R->L }
  implies(truth,implies(truth,implies(truth,implies(X,implies(X,Y)))))
= { by axiom 1 R->L }
  implies(truth,implies(truth,implies(truth,implies(truth,implies(X,implies(X,Y))))))
= { by axiom 2 }
  implies(truth,implies(truth,implies(implies(truth,implies(X,implies(X,Y))),truth)))
= { by axiom 2 R->L }
  implies(truth,implies(implies(implies(truth,implies(X,implies(X,Y))),truth),truth))
= { by axiom 3 }
  implies(truth,implies(implies(truth,implies(truth,implies(X,implies(X,Y)))),implies(truth,implies(X,implies(X,Y)))))
= { by axiom 2 }
  implies(truth,implies(implies(truth,implies(X,implies(X,Y))),implies(truth,implies(truth,implies(X,implies(X,Y))))))
= { by axiom 1 }
  implies(truth,implies(implies(truth,implies(X,implies(X,Y))),implies(truth,implies(X,implies(X,Y)))))
= { by axiom 1 R->L }
  implies(implies(truth,truth),implies(implies(truth,implies(X,implies(X,Y))),implies(truth,implies(X,implies(X,Y)))))
= { by axiom 4 }
  truth
