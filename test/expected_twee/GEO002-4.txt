Axiom 1: between(X,Y,extension(X,Y,Z,A))
Axiom 2: equidistant(B,extension(C,B,U,V),U,V)
Axiom 3: equidistant(C,B,W,W) => equalish(C,B)
Axiom 4: between(C,B,V) /\ between(B,X1,V) => between(C,B,X1)
Axiom 5: equalish(C,B) /\ between(U,X1,C) => between(U,X1,B)
Axiom 6: between(C,U,V) /\ between(B,V,X1) => between(X1,U,outer_pasch(U,C,B,X1,V))
Axiom 7: between(C,U,V) /\ between(B,V,X1) => between(C,outer_pasch(U,C,B,X1,V),B)

Lemma 8: between(a,a,a)
Proof:
  have between(a,a,extension(a,a,X,Y))
    by axiom 1
   and between(a,a,extension(a,a,X,Y))
    by axiom 1
  hence between(a,a,a)
    by axiom 4

Lemma 9: between(a,a,extension(b,a,X,X))
Proof:

   and between(a,a,a)
    by lemma 8
  hence between(a,a,extension(b,a,X,X))
