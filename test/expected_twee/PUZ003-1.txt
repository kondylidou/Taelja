Axiom 1: member(lorenzo)
Axiom 2: member(guido)
Axiom 3: member(cesare)
Axiom 4: shaved(guido,cesare)
Axiom 5: member(X) /\ member(Y) /\ shaved(X,Y) => shaved(members,X)
Axiom 6: shaved(members,Z) /\ member(A) => shaved(A,Z)
Axiom 7: member(petruchio)

Lemma 8: shaved(lorenzo,guido)
Proof:
  have member(guido)
    by axiom 2
   and member(cesare)
    by axiom 3
   and shaved(guido,cesare)
    by axiom 4
  hence shaved(members,guido)
    by axiom 5
   and member(lorenzo)
    by axiom 1
  hence shaved(lorenzo,guido)
    by axiom 6

Goal 1: shaved(petruchio,lorenzo)
Proof:
  have member(lorenzo)
    by axiom 1
   and member(guido)
    by axiom 2
   and shaved(lorenzo,guido)
    by lemma 8
  hence shaved(members,lorenzo)
    by axiom 5
   and member(petruchio)
    by axiom 7
  hence shaved(petruchio,lorenzo)
    by axiom 6
