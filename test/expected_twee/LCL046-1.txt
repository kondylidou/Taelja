Axiom 1: is_a_theorem(implies(X,implies(not(X),Y)))
Axiom 2: is_a_theorem(implies(implies(Z,A),implies(implies(A,B),implies(Z,B))))
Axiom 3: is_a_theorem(implies(Z,A)) /\ is_a_theorem(Z) => is_a_theorem(A)
Axiom 4: is_a_theorem(implies(implies(not(Z),Z),Z))

Lemma 5: is_a_theorem(implies(implies(implies(not(a),a),a),implies(a,a)))
Proof:
  have is_a_theorem(implies(implies(a,implies(not(a),a)),implies(implies(implies(not(a),a),a),implies(a,a))))
    by axiom 2
   and is_a_theorem(implies(a,implies(not(a),a)))
    by axiom 1
  hence is_a_theorem(implies(implies(implies(not(a),a),a),implies(a,a)))
    by axiom 3

Lemma 6: is_a_theorem(implies(implies(not(a),a),a))
Proof:
  have is_a_theorem(implies(implies(not(a),a),a))
    by axiom 4

Goal 1: is_a_theorem(implies(a,a))
Proof:
  have is_a_theorem(implies(implies(implies(not(a),a),a),implies(a,a)))
    by lemma 5
   and is_a_theorem(implies(implies(not(a),a),a))
    by lemma 6
  hence is_a_theorem(implies(a,a))
    by axiom 3
