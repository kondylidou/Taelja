Axiom 1: product(X,inverse(X),identity)
Axiom 2: product(Y,identity,Y)
Axiom 3: product(Y,Z,A) /\ product(Z,B,C) /\ product(Y,C,U) => product(A,B,U)
Axiom 4: product(identity,Y,Y)
Axiom 5: product(Y,V,W) /\ product(Y,V,X1) => W = X1

Lemma 6: product(inverse(X),inverse(inverse(X)),identity)
Proof:
  have product(inverse(X),inverse(inverse(X)),identity)
    by axiom 1

Lemma 7: product(X,identity,X)
Proof:
  have product(X,identity,X)
    by axiom 2

Lemma 8: product(identity,inverse(inverse(X)),inverse(inverse(X)))
Proof:
  have product(identity,inverse(inverse(X)),inverse(inverse(X)))
    by axiom 4

Goal 1: inverse(inverse(a)) = a
Proof:
  have product(a,inverse(a),identity)
    by axiom 1
   and product(inverse(a),inverse(inverse(a)),identity)
    by lemma 6
   and product(a,identity,a)
    by lemma 7
  hence product(identity,inverse(inverse(a)),a)
    by axiom 3
   and product(identity,inverse(inverse(a)),inverse(inverse(a)))
    by lemma 8
  hence inverse(inverse(a)) = a
    by axiom 5
