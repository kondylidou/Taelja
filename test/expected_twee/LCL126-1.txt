Axiom 1: is_a_theorem(equivalent(equivalent(X,Y),equivalent(equivalent(X,Z),equivalent(Y,Z))))
Axiom 2: is_a_theorem(equivalent(A,B)) /\ is_a_theorem(A) => is_a_theorem(B)

Lemma 3: is_a_theorem(equivalent(equivalent(equivalent(a,c),equivalent(a,c)),equivalent(equivalent(equivalent(a,b),equivalent(c,b)),equivalent(a,c))))
Proof:
  have is_a_theorem(equivalent(equivalent(equivalent(a,c),equivalent(equivalent(a,b),equivalent(c,b))),equivalent(equivalent(equivalent(a,c),equivalent(a,c)),equivalent(equivalent(equivalent(a,b),equivalent(c,b)),equivalent(a,c)))))
    by axiom 1
   and is_a_theorem(equivalent(equivalent(a,c),equivalent(equivalent(a,b),equivalent(c,b))))
    by axiom 1
  hence is_a_theorem(equivalent(equivalent(equivalent(a,c),equivalent(a,c)),equivalent(equivalent(equivalent(a,b),equivalent(c,b)),equivalent(a,c))))
    by axiom 2
