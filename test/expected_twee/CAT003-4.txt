Axiom 1: compose(X,compose(Y,Z)) = compose(compose(X,Y),Z)
Axiom 2: compose(h,a) = compose(g,a)
Axiom 3: compose(A,compose(a,b)) = B /\ compose(C,compose(a,b)) = B => A = C

Lemma 4: compose(g,compose(a,b)) = compose(compose(h,a),b)
Proof:
  have compose(g,compose(a,b)) = compose(compose(g,a),b)
    by axiom 1
  hence compose(g,compose(a,b)) = compose(compose(h,a),b)
    by rw axiom 2 R->L

Goal 1: g = h
Proof:
  have compose(h,compose(a,b)) = compose(compose(h,a),b)
    by axiom 1
   and compose(g,compose(a,b)) = compose(compose(h,a),b)
    by lemma 4
  hence g = h
    by axiom 3
