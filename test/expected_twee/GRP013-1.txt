Axiom 1: product(X,X,identity)
Axiom 2: product(inverse(a),inverse(b),d)
Axiom 3: product(Y,inverse(Y),identity)
Axiom 4: product(inverse(Z),inverse(A),B) => product(Z,B,A)
Axiom 5: product(Y,identity,Y)
Axiom 6: product(C,U,V) /\ product(C,U,W) => V = W
Axiom 7: product(a,b,c)

Lemma 8: product(X,identity,inverse(X))
Proof:
  have product(inverse(X),inverse(inverse(X)),identity)
    by axiom 3
  hence product(X,identity,inverse(X))
    by axiom 4

Lemma 9: inverse(X) = X
Proof:
  have product(X,identity,X)
    by axiom 5
   and product(X,identity,inverse(X))
    by lemma 8
  hence inverse(X) = X
    by axiom 6

Lemma 10: product(a,b,d)
Proof:
  have product(inverse(a),inverse(b),d)
    by axiom 2
  hence product(a,inverse(b),d)
    by rw lemma 9
  hence product(a,b,d)
    by rw lemma 9

Lemma 11: d = c
Proof:
  have product(a,b,c)
    by axiom 7
   and product(a,b,d)
    by lemma 10
  hence d = c
    by axiom 6

Goal 1: product(c,d,identity)
Proof:
  have product(c,c,identity)
    by axiom 1
  hence product(c,d,identity)
    by rw lemma 11 R->L
