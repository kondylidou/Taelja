Axiom 1: meet(zero,X) = zero
Axiom 2: meet(X,Y) = meet(Y,X)
Axiom 3: comp(b,join(c,d))
Axiom 4: comp(X,Y) => meet(X,Y) = zero
Axiom 5: meet(X,meet(Y,Z)) = meet(meet(X,Y),Z)
Axiom 6: join(A,B) = join(B,A)
Axiom 7: join(X,meet(X,Y)) = X
Axiom 8: join(X,join(Y,C)) = join(join(X,Y),C)
Axiom 9: meet(X,join(X,Y)) = X
Axiom 10: join(zero,X) = X

Lemma 11: join(meet(X,Y),Y) = Y
Proof:
  join(meet(X,Y),Y)
= { by axiom 6 }
  join(Y,meet(X,Y))
= { by axiom 2 }
  join(Y,meet(Y,X))
= { by axiom 7 }
  Y

Lemma 12: join(meet(X,Y),join(Z,Y)) = join(Z,Y)
Proof:
  join(meet(X,Y),join(Z,Y))
= { by axiom 6 }
  join(meet(X,Y),join(Y,Z))
= { by axiom 8 }
  join(join(meet(X,Y),Y),Z)
= { by lemma 11 }
  join(Y,Z)
= { by axiom 6 }
  join(Z,Y)

Lemma 13: meet(b,join(c,d)) = zero
Proof:
  have comp(b,join(c,d))
    by axiom 3
  hence meet(b,join(c,d)) = zero
    by axiom 4

Lemma 14: meet(b,c) = zero
Proof:
  meet(b,c)
= { by axiom 9 R->L }
  meet(meet(b,c),join(meet(b,c),join(d,c)))
= { by lemma 12 }
  meet(meet(b,c),join(d,c))
= { by axiom 6 }
  meet(meet(b,c),join(c,d))
= { by axiom 2 }
  meet(join(c,d),meet(b,c))
= { by axiom 2 R->L }
  meet(join(c,d),meet(c,b))
= { by axiom 2 R->L }
  meet(meet(c,b),join(c,d))
= { by axiom 5 R->L }
  meet(c,meet(b,join(c,d)))
= { by lemma 13 }
  meet(c,zero)
= { by axiom 2 }
  meet(zero,c)
= { by axiom 1 }
  zero

Lemma 15: join(meet(b,c),X) = X
Proof:
  join(meet(b,c),X)
= { by lemma 14 }
  join(zero,X)
= { by axiom 10 }
  X

Lemma 16: join(X,join(Y,Z)) = join(Z,join(Y,X))
Proof:
  join(X,join(Y,Z))
= { by axiom 6 R->L }
  join(X,join(Z,Y))
= { by axiom 6 R->L }
  join(join(Z,Y),X)
= { by axiom 8 R->L }
  join(Z,join(Y,X))

Goal 1: meet(join(a,meet(b,c)),join(a,meet(b,d))) = a
Proof:
  meet(join(a,meet(b,c)),join(a,meet(b,d)))
= { by lemma 15 R->L }
  meet(join(a,meet(b,c)),join(meet(b,c),join(a,meet(b,d))))
= { by lemma 16 }
  meet(join(a,meet(b,c)),join(meet(b,d),join(a,meet(b,c))))
= { by axiom 6 }
  meet(join(a,meet(b,c)),join(join(a,meet(b,c)),meet(b,d)))
= { by axiom 9 }
  join(a,meet(b,c))
= { by axiom 6 }
  join(meet(b,c),a)
= { by lemma 15 }
  a
