Axiom 1: join(X,Y) = join(Y,X)
Axiom 2: join(Z,join(A,B)) = join(join(Z,A),B)
Axiom 3: meet(Z,A) = meet(A,Z)
Axiom 4: join(Z,meet(Z,A)) = Z
Axiom 5: meet(Z,join(Z,A)) = Z
Axiom 6: meet(Z,meet(A,C)) = meet(meet(Z,A),C)
Axiom 7: comp(b,join(c,d))
Axiom 8: comp(Z,A) => meet(Z,A) = zero
Axiom 9: meet(zero,Z) = zero
Axiom 10: join(zero,Z) = Z

Lemma 11: meet(b,join(c,d)) = zero
Proof:
  have comp(b,join(c,d))
    by axiom 7
  hence meet(b,join(c,d)) = zero
    by axiom 8

Lemma 12: join(meet(b,c),X) = X
Proof:
  join(meet(b,c),X)
= { by axiom 5 R->L }
  join(meet(b,meet(c,join(c,d))),X)
= { by axiom 1 }
  join(meet(b,meet(c,join(d,c))),X)
= { by axiom 3 R->L }
  join(meet(b,meet(join(d,c),c)),X)
= { by axiom 6 }
  join(meet(meet(b,join(d,c)),c),X)
= { by axiom 1 R->L }
  join(meet(meet(b,join(c,d)),c),X)
= { by lemma 11 }
  join(meet(zero,c),X)
= { by axiom 9 }
  join(zero,X)
= { by axiom 10 }
  X

Goal 1: meet(join(a,meet(b,c)),join(a,meet(b,d))) = a
Proof:
  meet(join(a,meet(b,c)),join(a,meet(b,d)))
= { by lemma 12 R->L }
  meet(join(a,meet(b,c)),join(join(meet(b,c),a),meet(b,d)))
= { by axiom 1 R->L }
  meet(join(a,meet(b,c)),join(join(a,meet(b,c)),meet(b,d)))
= { by axiom 5 }
  join(a,meet(b,c))
= { by axiom 1 }
  join(meet(b,c),a)
= { by lemma 12 }
  a
