Axiom 1: join(X,Y) = join(Y,X)
Axiom 2: join(Z,join(A,B)) = join(join(Z,A),B)
Axiom 3: meet(Z,A) = meet(A,Z)
Axiom 4: join(Z,meet(Z,A)) = Z
Axiom 5: meet(Z,join(Z,A)) = Z
Axiom 6: meet(Z,meet(A,C)) = meet(meet(Z,A),C)
Axiom 7: comp(b,join(c,d))
Axiom 8: comp(Z,A) => meet(Z,A) = zero
Axiom 9: meet(zero,Z) = zero
Axiom 10: join(zero,Z) = Z

Lemma 11: meet(b,join(c,d)) = zero
Proof:
  have comp(b,join(c,d))
    by axiom 7
  hence meet(b,join(c,d)) = zero
    by axiom 8

Goal 1: meet(join(a,meet(b,c)),join(a,meet(b,d))) = a
Proof:

  join(meet(join(a,meet(b,c)),join(a,meet(b,d))),meet(meet(b,c),join(c,d)))

  join(meet(b,c),join(meet(join(a,meet(b,c)),join(a,meet(b,d))),meet(join(a,meet(b,c)),a)))
