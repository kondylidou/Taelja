Lemma 11: >=(+(X,Y),Y)
Proof:
  have >=(X,0)
    by axiom 5
  hence >=(+(X,Y),+(0,Y))
    by axiom 7
  hence >=(+(X,Y),+(Y,0))
    by rw axiom 3 R->L
  hence >=(+(X,Y),Y)
    by rw axiom 6

Lemma 12: +(1,X) = 1
Proof:

= { by axiom 3 R->L }
  +(X,1)
= { by axiom 4 }
  1

Lemma 13: >=(1,X)
Proof:
  have >=(+(1,X),X)

Lemma 14: >=(+(X,==>(X,Y)),Y)

Lemma 15: +(X,==>(X,1)) = 1

Lemma 16: >=(==>(X,Y),Z) /\ >=(+(X,Z),Y) => ==>(X,Y) = Z
