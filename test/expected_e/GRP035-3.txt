Axiom 1: subgroup_member(b)
Axiom 2: product(X,inverse(X),identity)
Axiom 3: subgroup_member(X) /\ subgroup_member(Y) /\ product(X,inverse(Y),Z) => subgroup_member(Z)
Axiom 4: product(identity,X,X)
Axiom 5: subgroup_member(a)
Axiom 6: product(X,Y,multiply(X,Y))
Axiom 7: product(a,b,c)
Axiom 8: product(X,Y,Z) /\ product(X,Y,A) => Z = A
Axiom 9: product(X,identity,X)
Axiom 10: product(X,Y,Z) /\ product(Y,A,B) /\ product(X,B,C) => product(Z,A,C)

Lemma 11: b = multiply(b,identity)
Proof:
  have product(b,identity,b)
    by axiom 9
   and product(b,identity,multiply(b,identity))
    by axiom 6
  hence b = multiply(b,identity)
    by axiom 8

Lemma 12: product(identity,inverse(inverse(b)),b)
Proof:
  have product(b,inverse(b),identity)
    by axiom 2
   and product(inverse(b),inverse(inverse(b)),identity)
    by axiom 2
   and product(b,identity,multiply(b,identity))
    by axiom 6
  hence product(identity,inverse(inverse(b)),multiply(b,identity))
    by axiom 10
  hence product(identity,inverse(inverse(b)),b)
    by rw lemma 11 R->L

Lemma 13: subgroup_member(inverse(b))
Proof:
  have subgroup_member(b)
    by axiom 1
   and subgroup_member(b)
    by axiom 1
   and product(b,inverse(b),identity)
    by axiom 2
  hence subgroup_member(identity)
    by axiom 3
   and subgroup_member(b)
    by axiom 1
   and product(identity,inverse(b),inverse(b))
    by axiom 4
  hence subgroup_member(inverse(b))
    by axiom 3

Lemma 14: inverse(inverse(b)) = b
Proof:
  have product(identity,inverse(inverse(b)),inverse(inverse(b)))
    by axiom 4
   and product(identity,inverse(inverse(b)),b)
    by lemma 12
  hence inverse(inverse(b)) = b
    by axiom 8

Lemma 15: c = multiply(a,b)
Proof:
  have product(a,b,c)
    by axiom 7
   and product(a,b,multiply(a,b))
    by axiom 6
  hence c = multiply(a,b)
    by axiom 8

Goal 1: subgroup_member(c)
Proof:
  have subgroup_member(a)
    by axiom 5
   and subgroup_member(inverse(b))
    by lemma 13
   and product(a,inverse(inverse(b)),multiply(a,inverse(inverse(b))))
    by axiom 6
  hence subgroup_member(multiply(a,inverse(inverse(b))))
    by axiom 3
  hence subgroup_member(multiply(a,b))
    by rw lemma 14
  hence subgroup_member(c)
    by rw lemma 15 R->L
