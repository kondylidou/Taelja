Axiom 1: product(X,Y,multiply(X,Y))
Axiom 2: product(identity,X,X)
Axiom 3: product(X,Y,Z) /\ product(X,Y,A) => Z = A
Axiom 4: product(a,b,c)
Axiom 5: product(inverse(b),inverse(a),d)
Axiom 6: product(X,inverse(X),identity)
Axiom 7: product(X,Y,Z) /\ product(Y,A,B) /\ product(X,B,C) => product(Z,A,C)
Axiom 8: product(X,identity,X)
Axiom 9: product(inverse(X),X,identity)

Lemma 10: multiply(identity,X) = X
Proof:
  have product(identity,X,X)
    by axiom 2
   and product(identity,X,multiply(identity,X))
    by axiom 1
  hence multiply(identity,X) = X
    by axiom 3

Lemma 11: product(identity,inverse(a),multiply(b,d))
Proof:
  have product(b,inverse(b),identity)
    by axiom 6
   and product(inverse(b),inverse(a),d)
    by axiom 5
   and product(b,d,multiply(b,d))
    by axiom 1
  hence product(identity,inverse(a),multiply(b,d))
    by axiom 7

Lemma 12: multiply(identity,inverse(a)) = multiply(b,d)
Proof:
  have product(identity,inverse(a),multiply(identity,inverse(a)))
    by axiom 1
   and product(identity,inverse(a),multiply(b,d))
    by lemma 11
  hence multiply(identity,inverse(a)) = multiply(b,d)
    by axiom 3

Lemma 13: product(b,d,inverse(a))
Proof:
  product(b,d,inverse(a))
= { by lemma 10 R->L }
  product(b,d,multiply(identity,inverse(a)))
= { by lemma 12 }
  product(b,d,multiply(b,d))
= { by axiom 1 }
  true

Lemma 14: c = multiply(a,b)
Proof:
  have product(a,b,c)
    by axiom 4
   and product(a,b,multiply(a,b))
    by axiom 1
  hence c = multiply(a,b)
    by axiom 3

Lemma 15: product(c,d,identity)
Proof:
  have product(a,b,multiply(a,b))
    by axiom 1
   and product(b,d,inverse(a))
    by lemma 13
   and product(a,inverse(a),identity)
    by axiom 6
  hence product(multiply(a,b),d,identity)
    by axiom 7
  hence product(c,d,identity)
    by rw lemma 14 R->L

Lemma 16: product(identity,d,inverse(c))
Proof:
  have product(inverse(c),c,identity)
    by axiom 9
   and product(c,d,identity)
    by lemma 15
   and product(inverse(c),identity,inverse(c))
    by axiom 8
  hence product(identity,d,inverse(c))
    by axiom 7

Goal 1: inverse(c) = d
Proof:
  have product(identity,d,multiply(identity,d))
    by axiom 1
   and product(identity,d,inverse(c))
    by lemma 16
  hence multiply(identity,d) = inverse(c)
    by axiom 3
  hence d = inverse(c)
    by rw lemma 10
