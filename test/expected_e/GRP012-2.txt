Axiom 1: product(X,Y,multiply(X,Y))
Axiom 2: product(inverse(X),X,identity)
Axiom 3: product(a,b,c)
Axiom 4: product(X,Y,Z) /\ product(Y,A,B) /\ product(X,B,C) => product(Z,A,C)
Axiom 5: product(X,identity,X)
Axiom 6: product(X,Y,Z) /\ product(X,Y,A) => Z = A
Axiom 7: product(X,inverse(X),identity)
Axiom 8: product(identity,X,X)
Axiom 9: product(inverse(b),inverse(a),d)

Lemma 10: X = multiply(X,identity)
Proof:
  have product(X,identity,X)
    by axiom 5
   and product(X,identity,multiply(X,identity))
    by axiom 1
  hence X = multiply(X,identity)
    by axiom 6

Lemma 11: product(X,Y,Z) => product(Z,inverse(Y),X)
Proof:
  have product(X,Y,Z)
    by assumption
   and product(Y,inverse(Y),identity)
    by axiom 7
   and product(X,identity,multiply(X,identity))
    by axiom 1
  hence product(Z,inverse(Y),multiply(X,identity))
    by axiom 4
  hence product(Z,inverse(Y),X)
    by rw lemma 10 R->L

Lemma 12: product(identity,inverse(b),multiply(inverse(c),a))
Proof:
  have product(inverse(c),a,multiply(inverse(c),a))
    by axiom 1
   and product(a,b,c)
    by axiom 3
   and product(inverse(c),c,identity)
    by axiom 2
  hence product(multiply(inverse(c),a),b,identity)
    by axiom 4
  hence product(identity,inverse(b),multiply(inverse(c),a))
    by lemma 11

Lemma 13: inverse(b) = multiply(inverse(c),a)
Proof:
  have product(identity,inverse(b),inverse(b))
    by axiom 8
   and product(identity,inverse(b),multiply(inverse(c),a))
    by lemma 12
  hence inverse(b) = multiply(inverse(c),a)
    by axiom 6

Lemma 14: product(inverse(b),inverse(a),inverse(c))
Proof:
  have product(inverse(c),a,multiply(inverse(c),a))
    by axiom 1
  hence product(inverse(c),a,inverse(b))
    by rw lemma 13 R->L
  hence product(inverse(b),inverse(a),inverse(c))
    by lemma 11

Goal 1: inverse(c) = d
Proof:
  have product(inverse(b),inverse(a),d)
    by axiom 9
   and product(inverse(b),inverse(a),inverse(c))
    by lemma 14
  hence inverse(c) = d
    by axiom 6
