Axiom 1: product(X,Y,multiply(X,Y))
Axiom 2: product(identity,X,X)
Axiom 3: product(X,Y,Z) /\ product(X,Y,A) => Z = A
Axiom 4: product(X,inverse(X),identity)
Axiom 5: product(X,identity,X)
Axiom 6: product(c,b,identity)
Axiom 7: product(X,Y,Z) /\ product(Y,A,B) /\ product(X,B,C) => product(Z,A,C)
Axiom 8: product(a,b,identity)
Axiom 9: product(inverse(X),X,identity)

Lemma 10: multiply(identity,X) = X
Proof:
  have product(identity,X,multiply(identity,X))
    by axiom 1
   and product(identity,X,X)
    by axiom 2
  hence multiply(identity,X) = X
    by axiom 3

Lemma 11: c = multiply(identity,inverse(b))
Proof:
  have product(c,b,identity)
    by axiom 6
  hence product(c,b,multiply(identity,identity))
    by rw lemma 10 R->L
   and product(b,inverse(b),identity)
    by axiom 4
   and product(c,identity,c)
    by axiom 5
  hence product(multiply(identity,identity),inverse(b),c)
    by axiom 7
  hence product(identity,inverse(b),c)
    by rw lemma 10
   and product(identity,inverse(b),multiply(identity,inverse(b)))
    by axiom 1
  hence c = multiply(identity,inverse(b))
    by axiom 3

Lemma 12: inverse(a) = multiply(identity,b)
Proof:
  have product(inverse(a),a,identity)
    by axiom 9
   and product(a,b,identity)
    by axiom 8
   and product(inverse(a),identity,inverse(a))
    by axiom 5
  hence product(identity,b,inverse(a))
    by axiom 7
   and product(identity,b,multiply(identity,b))
    by axiom 1
  hence inverse(a) = multiply(identity,b)
    by axiom 3

Lemma 13: product(identity,X,inverse(inverse(X)))
Proof:
  have product(inverse(Y),Y,identity)
    by axiom 9
   and product(inverse(X),X,identity)
    by axiom 9
   and product(inverse(inverse(X)),identity,inverse(inverse(X)))
    by axiom 5
  hence product(identity,X,inverse(inverse(X)))
    by axiom 7

Lemma 14: product(identity,a,c)
Proof:
  product(identity,a,c)
= { by lemma 11 }
  product(identity,a,multiply(identity,inverse(b)))
= { by lemma 10 }
  product(identity,a,inverse(b))
= { by lemma 10 R->L }
  product(identity,a,inverse(multiply(identity,b)))
= { by lemma 12 R->L }
  product(identity,a,inverse(inverse(a)))
= { by lemma 13 }
  true

Goal 1: a = c
Proof:
  have product(identity,a,a)
    by axiom 2
   and product(identity,a,c)
    by lemma 14
  hence a = c
    by axiom 3
