Axiom 1: product(X,Y,multiply(X,Y))
Axiom 2: product(identity,X,X)
Axiom 3: product(X,Y,Z) /\ product(X,Y,A) => Z = A
Axiom 4: product(X,inverse(X),identity)
Axiom 5: product(X,identity,X)
Axiom 6: product(c,b,identity)
Axiom 7: product(X,Y,Z) /\ product(Y,A,B) /\ product(X,B,C) => product(Z,A,C)
Axiom 8: product(a,b,identity)
Axiom 9: product(inverse(X),X,identity)

Lemma 10: multiply(identity,X) = X
Proof:
  have product(identity,X,X)
    by axiom 2
   and product(identity,X,multiply(identity,X))
    by axiom 1
  hence multiply(identity,X) = X
    by axiom 3

Lemma 11: product(identity,inverse(b),c)
Proof:
  have product(c,b,identity)
    by axiom 6
   and product(b,inverse(b),identity)
    by axiom 4
   and product(c,identity,c)
    by axiom 5
  hence product(identity,inverse(b),c)
    by axiom 7

Lemma 12: product(identity,b,inverse(a))
Proof:
  have product(inverse(a),a,identity)
    by axiom 9
   and product(a,b,identity)
    by axiom 8
   and product(inverse(a),identity,inverse(a))
    by axiom 5
  hence product(identity,b,inverse(a))
    by axiom 7

Lemma 13: multiply(identity,b) = inverse(a)
Proof:
  have product(identity,b,multiply(identity,b))
    by axiom 1
   and product(identity,b,inverse(a))
    by lemma 12
  hence multiply(identity,b) = inverse(a)
    by axiom 3

Lemma 14: product(b,a,identity)
Proof:
  product(b,a,identity)
= { by lemma 10 R->L }
  product(multiply(identity,b),a,identity)
= { by lemma 13 }
  product(inverse(a),a,identity)
= { by axiom 9 }
  true

Lemma 15: multiply(identity,inverse(b)) = c
Proof:
  have product(identity,inverse(b),multiply(identity,inverse(b)))
    by axiom 1
   and product(identity,inverse(b),c)
    by lemma 11
  hence multiply(identity,inverse(b)) = c
    by axiom 3

Lemma 16: product(identity,a,inverse(b))
Proof:
  have product(inverse(b),b,identity)
    by axiom 9
   and product(b,a,identity)
    by lemma 14
   and product(inverse(b),identity,inverse(b))
    by axiom 5
  hence product(identity,a,inverse(b))
    by axiom 7

Lemma 17: product(identity,a,c)
Proof:
  product(identity,a,c)
= { by lemma 15 R->L }
  product(identity,a,multiply(identity,inverse(b)))
= { by lemma 10 }
  product(identity,a,inverse(b))
= { by lemma 16 }
  true

Goal 1: a = c
Proof:
  have product(identity,a,multiply(identity,a))
    by axiom 1
   and product(identity,a,c)
    by lemma 17
  hence multiply(identity,a) = c
    by axiom 3
  hence a = c
    by rw lemma 10
