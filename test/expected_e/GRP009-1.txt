Axiom 1: product(X,Y,multiply(X,Y))
Axiom 2: product(identity,X,X)
Axiom 3: product(X,Y,Z) /\ product(X,Y,A) => Z = A
Axiom 4: product(inverse(X),X,identity)
Axiom 5: product(X,identity,X)
Axiom 6: product(a,b,identity)
Axiom 7: product(X,Y,Z) /\ product(Y,A,B) /\ product(X,B,C) => product(Z,A,C)
Axiom 8: product(c,b,identity)
Axiom 9: product(X,Y,Z) /\ product(Y,A,B) /\ product(Z,A,C) => product(X,B,C)

Lemma 10: product(X,identity,Y) => Y = X
Proof:
  have product(X,identity,X)
    by axiom 5
   and product(X,identity,Y)
    by assumption
  hence Y = X
    by axiom 3

Lemma 11: multiply(inverse(a),identity) = inverse(a)
Proof:
  have product(inverse(a),identity,multiply(inverse(a),identity))
    by axiom 1
  hence multiply(inverse(a),identity) = inverse(a)
    by lemma 10

Lemma 12: product(identity,b,inverse(a))
Proof:
  have product(inverse(a),a,identity)
    by axiom 4
   and product(a,b,identity)
    by axiom 6
   and product(inverse(a),identity,multiply(inverse(a),identity))
    by axiom 1
  hence product(identity,b,multiply(inverse(a),identity))
    by axiom 7
  hence product(identity,b,inverse(a))
    by rw lemma 11

Lemma 13: multiply(identity,b) = inverse(a)
Proof:
  have product(identity,b,multiply(identity,b))
    by axiom 1
   and product(identity,b,inverse(a))
    by lemma 12
  hence multiply(identity,b) = inverse(a)
    by axiom 3

Lemma 14: b = multiply(identity,b)
Proof:
  have product(identity,b,b)
    by axiom 2
   and product(identity,b,multiply(identity,b))
    by axiom 1
  hence b = multiply(identity,b)
    by axiom 3

Lemma 15: inverse(a) = b
Proof:
  inverse(a)
= { by lemma 13 R->L }
  multiply(identity,b)
= { by lemma 14 R->L }
  b

Lemma 16: product(b,a,identity)
Proof:
  have product(inverse(a),a,identity)
    by axiom 4
  hence product(b,a,identity)
    by rw lemma 15

Goal 1: a = c
Proof:
  have product(c,b,identity)
    by axiom 8
   and product(b,a,identity)
    by lemma 16
   and product(identity,a,a)
    by axiom 2
  hence product(c,identity,a)
    by axiom 9
  hence a = c
    by lemma 10
