Axiom 1: ld(X,mult(X,Y)) = Y
Axiom 2: mult(rd(X,Y),Y) = X
Axiom 3: mult(mult(X,Y),mult(Z,X)) = mult(mult(X,mult(Y,Z)),X)
Axiom 4: rd(mult(X,Y),Y) = X

Lemma 5: mult(X,mult(Y,rd(X,X))) = mult(X,Y)
Proof:
  mult(X,mult(Y,rd(X,X)))
= { by axiom 4 R->L }
  rd(mult(mult(X,mult(Y,rd(X,X))),X),X)
= { by axiom 3 R->L }
  rd(mult(mult(X,Y),mult(rd(X,X),X)),X)
= { by axiom 2 }
  rd(mult(mult(X,Y),X),X)
= { by axiom 4 }
  mult(X,Y)

Lemma 6: mult(X,rd(Y,Y)) = X
Proof:
  mult(X,rd(Y,Y))
= { by axiom 1 R->L }
  ld(Y,mult(Y,mult(X,rd(Y,Y))))
= { by lemma 5 }
  ld(Y,mult(Y,X))
= { by axiom 1 }
  X

Lemma 7: ld(X,X) = rd(Y,Y)
Proof:
  ld(X,X)
= { by lemma 6 R->L }
  ld(X,mult(X,rd(Y,Y)))
= { by axiom 1 }
  rd(Y,Y)

Goal 1: mult(X,rd(Y,Y)) = X
Proof:
  mult(X,rd(Y,Y))
= { by lemma 6 }
  X

Goal 2: mult(rd(X,X),Y) = Y
Proof:
  mult(rd(X,X),Y)
= { by lemma 7 R->L }
  mult(ld(X,X),Y)
= { by lemma 7 }
  mult(rd(Y,Y),Y)
= { by axiom 2 }
  Y
