Axiom 1: labels(loop,p3)
Axiom 2: has(p8,goto(loop))
Axiom 3: has(X,goto(Y)) /\ labels(Y,Z) => succeeds(Z,X)
Axiom 4: follows(p8,p3)
Axiom 5: follows(Z,X) => succeeds(Z,X)
Axiom 6: succeeds(Z,A) /\ succeeds(A,X) => succeeds(Z,X)

Lemma 7: succeeds(p8,p3)
Proof:
  have follows(p8,p3)
    by axiom 4
  hence succeeds(p8,p3)
    by axiom 5

Goal 1: succeeds(p3,p3)
Proof:
  have has(p8,goto(loop))
    by axiom 2
   and labels(loop,p3)
    by axiom 1
  hence succeeds(p3,p8)
    by axiom 3
   and succeeds(p8,p3)
    by lemma 7
  hence succeeds(p3,p3)
    by axiom 6
