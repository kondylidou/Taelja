Axiom 1: least_upper_bound(identity,X) = X
Axiom 2: greatest_lower_bound(X,least_upper_bound(X,Y)) = X
Axiom 3: greatest_lower_bound(X,Y) = greatest_lower_bound(Y,X)
Axiom 4: multiply(inverse(X),X) = identity
Axiom 5: multiply(identity,X) = X
Axiom 6: multiply(multiply(X,Y),Z) = multiply(X,multiply(Y,Z))
Axiom 7: multiply(X,greatest_lower_bound(Y,Z)) = greatest_lower_bound(multiply(X,Y),multiply(X,Z))

Lemma 8: greatest_lower_bound(X,identity) = identity
Proof:
  greatest_lower_bound(X,identity)
= { by axiom 3 }
  greatest_lower_bound(identity,X)
= { by axiom 1 R->L }
  greatest_lower_bound(identity,least_upper_bound(identity,X))
= { by axiom 2 }
  identity

Lemma 9: multiply(inverse(X),multiply(X,Y)) = Y
Proof:
  multiply(inverse(X),multiply(X,Y))
= { by axiom 6 R->L }
  multiply(multiply(inverse(X),X),Y)
= { by axiom 4 }
  multiply(identity,Y)
= { by axiom 5 }
  Y

Lemma 10: multiply(inverse(inverse(X)),identity) = X
Proof:
  multiply(inverse(inverse(X)),identity)
= { by axiom 4 R->L }
  multiply(inverse(inverse(X)),multiply(inverse(X),X))
= { by lemma 9 }
  X

Lemma 11: multiply(inverse(inverse(inverse(X))),X) = identity
Proof:
  multiply(inverse(inverse(inverse(X))),X)
= { by lemma 10 R->L }
  multiply(inverse(inverse(inverse(X))),multiply(inverse(inverse(X)),identity))
= { by lemma 9 }
  identity

Lemma 12: inverse(inverse(X)) = X
Proof:
  inverse(inverse(X))
= { by lemma 10 R->L }
  multiply(inverse(inverse(inverse(inverse(X)))),identity)
= { by lemma 11 R->L }
  multiply(inverse(inverse(inverse(inverse(X)))),multiply(inverse(inverse(inverse(X))),X))
= { by lemma 9 }
  X

Lemma 13: multiply(X,identity) = X
Proof:
  multiply(X,identity)
= { by lemma 12 R->L }
  multiply(inverse(inverse(X)),identity)
= { by lemma 10 }
  X

Lemma 14: multiply(X,inverse(X)) = identity
Proof:
  multiply(X,inverse(X))
= { by lemma 12 R->L }
  multiply(inverse(inverse(X)),inverse(X))
= { by axiom 4 }
  identity

Lemma 15: identity = X
Proof:
  identity
= { by lemma 8 R->L }
  greatest_lower_bound(X,identity)
= { by lemma 14 R->L }
  greatest_lower_bound(X,multiply(X,inverse(X)))
= { by axiom 3 R->L }
  greatest_lower_bound(multiply(X,inverse(X)),X)
= { by lemma 13 R->L }
  greatest_lower_bound(multiply(X,inverse(X)),multiply(X,identity))
= { by axiom 7 R->L }
  multiply(X,greatest_lower_bound(inverse(X),identity))
= { by lemma 8 }
  multiply(X,identity)
= { by lemma 13 }
  X

Goal 1: multiply(a,b) = multiply(b,a)
Proof:
  multiply(a,b)
= { by lemma 15 R->L }
  identity
= { by lemma 15 }
  multiply(b,a)
