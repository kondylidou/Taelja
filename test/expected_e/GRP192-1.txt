Axiom 1: multiply(inverse(X),X) = identity
Axiom 2: multiply(identity,X) = X
Axiom 3: multiply(multiply(X,Y),Z) = multiply(X,multiply(Y,Z))
Axiom 4: greatest_lower_bound(X,Y) = greatest_lower_bound(Y,X)
Axiom 5: least_upper_bound(identity,X) = X
Axiom 6: greatest_lower_bound(X,least_upper_bound(X,Y)) = X
Axiom 7: multiply(X,greatest_lower_bound(Y,Z)) = greatest_lower_bound(multiply(X,Y),multiply(X,Z))

Lemma 8: multiply(inverse(X),multiply(X,Y)) = Y
Proof:
  multiply(inverse(X),multiply(X,Y))
= { by axiom 3 R->L }
  multiply(multiply(inverse(X),X),Y)
= { by axiom 1 }
  multiply(identity,Y)
= { by axiom 2 }
  Y

Lemma 9: multiply(inverse(inverse(X)),Y) = multiply(X,Y)
Proof:
  multiply(inverse(inverse(X)),Y)
= { by lemma 8 R->L }
  multiply(inverse(inverse(X)),multiply(inverse(X),multiply(X,Y)))
= { by lemma 8 }
  multiply(X,Y)

Lemma 10: multiply(X,identity) = X
Proof:
  multiply(X,identity)
= { by lemma 9 R->L }
  multiply(inverse(inverse(X)),identity)
= { by axiom 1 R->L }
  multiply(inverse(inverse(X)),multiply(inverse(X),X))
= { by lemma 8 }
  X

Lemma 11: inverse(inverse(X)) = X
Proof:
  inverse(inverse(X))
= { by lemma 10 R->L }
  multiply(inverse(inverse(X)),identity)
= { by lemma 9 }
  multiply(X,identity)
= { by lemma 10 }
  X

Lemma 12: multiply(X,multiply(inverse(X),Y)) = Y
Proof:
  multiply(X,multiply(inverse(X),Y))
= { by lemma 11 R->L }
  multiply(inverse(inverse(X)),multiply(inverse(X),Y))
= { by lemma 8 }
  Y

Lemma 13: greatest_lower_bound(X,identity) = identity
Proof:
  greatest_lower_bound(X,identity)
= { by axiom 4 R->L }
  greatest_lower_bound(identity,X)
= { by axiom 5 R->L }
  greatest_lower_bound(identity,least_upper_bound(identity,X))
= { by axiom 6 }
  identity

Lemma 14: greatest_lower_bound(X,Y) = X
Proof:
  greatest_lower_bound(X,Y)
= { by lemma 12 R->L }
  greatest_lower_bound(X,multiply(X,multiply(inverse(X),Y)))
= { by axiom 4 R->L }
  greatest_lower_bound(multiply(X,multiply(inverse(X),Y)),X)
= { by lemma 10 R->L }
  greatest_lower_bound(multiply(X,multiply(inverse(X),Y)),multiply(X,identity))
= { by axiom 7 R->L }
  multiply(X,greatest_lower_bound(multiply(inverse(X),Y),identity))
= { by lemma 13 }
  multiply(X,identity)
= { by lemma 10 }
  X

Lemma 15: X = Y
Proof:
  X
= { by lemma 14 R->L }
  greatest_lower_bound(X,Y)
= { by axiom 4 R->L }
  greatest_lower_bound(Y,X)
= { by lemma 14 }
  Y

Goal 1: multiply(a,b) = multiply(b,a)

  multiply(a,b)
= { by lemma 15 }
  multiply(b,a)
