  r(v(b,b,a,a),a,u(a,a,b))
= { by axiom 1 R->L }
  r(v(b,b,a,a),a,m(a,a,u(a,a,b)))
= { by axiom 1 R->L }
  r(v(b,b,a,a),m(u(a,a,b),u(a,a,b),a),m(a,a,u(a,a,b)))
= { by lemma 46 }
  true

Lemma 48: r(v(b,b,a,a),a,a)
Proof:
  have r(v(b,b,a,a),a,u(a,a,b))
    by lemma 47
   and r(u(a,b,b),u(a,a,b),u(a,a,b))
    by lemma 38
   and r(u(a,b,b),u(a,a,b),a)
    by lemma 20
  hence r(m(v(b,b,a,a),u(a,b,b),u(a,b,b)),m(a,u(a,a,b),u(a,a,b)),m(u(a,a,b),u(a,a,b),a))
    by axiom 15
  hence r(v(b,b,a,a),m(a,u(a,a,b),u(a,a,b)),m(u(a,a,b),u(a,a,b),a))
    by rw axiom 3
  hence r(v(b,b,a,a),a,m(u(a,a,b),u(a,a,b),a))
    by rw axiom 3
  hence r(v(b,b,a,a),a,a)
    by rw axiom 1

Lemma 49: r(b,b,a)
Proof:
  have r(v(b,b,a,a),u(a,b,b),a)
    by lemma 40
   and r(a,u(a,b,b),a)
    by lemma 41
   and r(a,b,a)
    by axiom 11
  hence r(m(v(b,b,a,a),a,a),m(u(a,b,b),u(a,b,b),b),m(a,a,a))
    by axiom 15
  hence r(v(b,b,a,a),m(u(a,b,b),u(a,b,b),b),m(a,a,a))
    by rw axiom 3
  hence r(v(b,b,a,a),m(u(a,b,b),u(a,b,b),b),a)
    by rw axiom 1
  hence r(v(b,b,a,a),b,a)
    by rw axiom 1
   and r(v(b,b,a,a),a,a)
    by lemma 48
   and r(b,a,a)
    by axiom 2
  hence r(m(v(b,b,a,a),v(b,b,a,a),b),m(b,a,a),m(a,a,a))
    by axiom 15
  hence r(m(v(b,b,a,a),v(b,b,a,a),b),b,m(a,a,a))
    by rw axiom 3
  hence r(m(v(b,b,a,a),v(b,b,a,a),b),b,a)
    by rw axiom 1
  hence r(b,b,a)
    by rw axiom 1

Goal 1: r(a,a,a)
Proof:
  have r(b,a,a)
    by axiom 2
   and r(b,b,a)
    by lemma 49
   and r(a,b,a)
    by axiom 11
  hence r(m(b,b,a),m(a,b,b),m(a,a,a))
    by axiom 15
  hence r(m(b,b,a),a,m(a,a,a))
    by rw axiom 3
  hence r(m(b,b,a),a,a)
    by rw axiom 1
  hence r(a,a,a)
    by rw axiom 1
