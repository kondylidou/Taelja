Axiom 1: negate(add(negate(e),negate(add(d,negate(e))))) = d
Axiom 2: add(X,Y) = add(Y,X)
Axiom 3: add(add(X,Y),Z) = add(X,add(Y,Z))
Axiom 4: negate(add(negate(add(X,Y)),negate(add(X,negate(Y))))) = X
Axiom 5: negate(add(X,negate(add(Y,Z)))) = negate(add(Y,negate(add(X,Z)))) => X = Y
Axiom 6: multiply(one,X) = X

Lemma 7: negate(add(negate(add(X,Y)),negate(add(negate(Y),X)))) = X
Proof:
  negate(add(negate(add(X,Y)),negate(add(negate(Y),X))))
= { by axiom 2 R->L }
  negate(add(negate(add(X,Y)),negate(add(X,negate(Y)))))
= { by axiom 4 }
  X

Lemma 8: negate(add(negate(add(X,Y)),negate(add(Y,negate(X))))) = Y
Proof:
  negate(add(negate(add(X,Y)),negate(add(Y,negate(X)))))
= { by axiom 2 R->L }
  negate(add(negate(add(Y,X)),negate(add(Y,negate(X)))))
= { by axiom 4 }
  Y

Lemma 9: add(X,add(Y,Z)) = add(Y,add(X,Z))
Proof:
  add(X,add(Y,Z))
= { by axiom 3 R->L }
  add(add(X,Y),Z)
= { by axiom 2 }
  add(add(Y,X),Z)
= { by axiom 3 }
  add(Y,add(X,Z))

Lemma 10: negate(add(d,negate(add(e,negate(add(d,negate(e))))))) = negate(add(d,negate(e)))
Proof:
  negate(add(d,negate(add(e,negate(add(d,negate(e)))))))
= { by axiom 2 R->L }
  negate(add(negate(add(e,negate(add(d,negate(e))))),d))
= { by axiom 2 R->L }
  negate(add(negate(add(negate(add(d,negate(e))),e)),d))
= { by axiom 1 R->L }
  negate(add(negate(add(negate(add(d,negate(e))),e)),negate(add(negate(e),negate(add(d,negate(e)))))))
= { by lemma 7 }
  negate(add(d,negate(e)))

Lemma 11: negate(add(negate(e),negate(add(negate(add(e,add(d,negate(add(d,negate(e)))))),d)))) = negate(add(negate(add(e,add(d,negate(add(d,negate(e)))))),negate(add(negate(e),d))))
Proof:
  negate(add(negate(e),negate(add(negate(add(e,add(d,negate(add(d,negate(e)))))),d))))
= { by axiom 3 R->L }
  negate(add(negate(e),negate(add(negate(add(add(e,d),negate(add(d,negate(e))))),d))))
= { by axiom 2 R->L }
  negate(add(negate(e),negate(add(negate(add(negate(add(d,negate(e))),add(e,d))),d))))
= { by lemma 8 R->L }
  negate(add(negate(e),negate(add(negate(add(negate(add(d,negate(e))),add(e,d))),negate(add(negate(add(e,d)),negate(add(d,negate(e)))))))))
= { by lemma 7 }
  negate(add(negate(e),negate(add(d,negate(e)))))
= { by axiom 1 }
  d
= { by axiom 4 R->L }
  negate(add(negate(add(d,add(e,negate(add(d,negate(e)))))),negate(add(d,negate(add(e,negate(add(d,negate(e)))))))))
= { by lemma 10 }
  negate(add(negate(add(d,add(e,negate(add(d,negate(e)))))),negate(add(d,negate(e)))))
= { by lemma 9 }
  negate(add(negate(add(e,add(d,negate(add(d,negate(e)))))),negate(add(d,negate(e)))))
= { by axiom 2 R->L }
  negate(add(negate(add(e,add(d,negate(add(d,negate(e)))))),negate(add(negate(e),d))))

Goal 1: negate(add(e,multiply(one,add(d,negate(add(d,negate(e))))))) = negate(e)
Proof:
  have negate(add(negate(e),negate(add(negate(add(e,add(d,negate(add(d,negate(e)))))),d)))) = negate(add(negate(add(e,add(d,negate(add(d,negate(e)))))),negate(add(negate(e),d))))
    by lemma 11
  hence negate(e) = negate(add(e,add(d,negate(add(d,negate(e))))))
    by axiom 5
  hence negate(e) = negate(add(e,multiply(one,add(d,negate(add(d,negate(e)))))))
    by rw axiom 6 R->L
